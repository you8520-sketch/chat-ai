import { createRequire } from "node:module";
import fs from "node:fs";
import path from "node:path";

const sharp = createRequire("/workspace/package.json")("sharp") as typeof import("sharp");

const OUT_DIR = "/opt/cursor/artifacts";
const DOC_DIR = "/workspace/docs/official-supply/reviews/assets/wolfgang-sig4-style-fidelity";
const SIZE = 400;

type Crop = { left: number; top: number };

const sources = {
  b13ref: {
    file: "/workspace/public/official-supply/style-seeds/romance-fantasy-cluster-b-v1/primary/b13-black-red-fur.webp",
    label: "b13 STYLE ref",
    fur: { left: 20, top: 380 },
    hair: { left: 360, top: 20 },
    metal: { left: 300, top: 780 },
  },
  candB7: {
    file: "/workspace/docs/official-supply/reviews/assets/wolfgang-sig4-style-fidelity/sig4-candidate.webp",
    label: "candidate b7 STYLE",
    fur: { left: 0, top: 160 },
    hair: { left: 820, top: 0 },
    metal: { left: 640, top: 400 },
  },
  candB13: {
    file: "/workspace/docs/official-supply/reviews/assets/wolfgang-sig4-style-fidelity/sig4-candidate-b13.webp",
    label: "candidate b13 STYLE",
    fur: { left: 40, top: 200 },
    hair: { left: 780, top: 0 },
    metal: { left: 520, top: 380 },
  },
} as const;

async function labeledCrop(file: string, crop: Crop, label: string): Promise<Buffer> {
  const meta = await sharp(file, { failOn: "none" }).metadata();
  const width = meta.width ?? SIZE;
  const height = meta.height ?? SIZE;
  const left = Math.min(crop.left, Math.max(0, width - SIZE));
  const top = Math.min(crop.top, Math.max(0, height - SIZE));
  const extractW = Math.min(SIZE, width - left);
  const extractH = Math.min(SIZE, height - top);
  const cropBuf = await sharp(file, { failOn: "none" })
    .extract({ left, top, width: extractW, height: extractH })
    .resize(SIZE, SIZE, { fit: "fill" })
    .png()
    .toBuffer();
  const svg = Buffer.from(
    `<svg width="${SIZE}" height="36"><rect width="100%" height="100%" fill="#111"/><text x="12" y="24" font-size="16" fill="#fff" font-family="sans-serif">${label}</text></svg>`
  );
  return sharp({
    create: { width: SIZE, height: SIZE + 36, channels: 3, background: "#111" },
  })
    .composite([
      { input: cropBuf, top: 36, left: 0 },
      { input: svg, top: 0, left: 0 },
    ])
    .png()
    .toBuffer();
}

async function labeledThumb(file: string, label: string, boxW = 420, boxH = 420): Promise<Buffer> {
  const resized = await sharp(file, { failOn: "none" })
    .resize(boxW, boxH, { fit: "contain", background: "#000" })
    .png()
    .toBuffer();
  const svg = Buffer.from(
    `<svg width="${boxW}" height="36"><rect width="100%" height="100%" fill="#111"/><text x="12" y="24" font-size="16" fill="#fff" font-family="sans-serif">${label}</text></svg>`
  );
  return sharp({
    create: { width: boxW, height: boxH + 36, channels: 3, background: "#000" },
  })
    .composite([
      { input: resized, top: 36, left: 0 },
      { input: svg, top: 0, left: 0 },
    ])
    .png()
    .toBuffer();
}

async function row(buffers: Buffer[], gap = 8): Promise<Buffer> {
  const metas = await Promise.all(buffers.map((buf) => sharp(buf).metadata()));
  const height = Math.max(...metas.map((meta) => meta.height ?? 0));
  const width = metas.reduce((sum, meta) => sum + (meta.width ?? 0), 0) + gap * (buffers.length - 1);
  let left = 0;
  const composites = buffers.map((input, index) => {
    const item = { input, top: 0, left };
    left += (metas[index]?.width ?? 0) + gap;
    return item;
  });
  return sharp({ create: { width, height, channels: 3, background: "#000" } })
    .composite(composites)
    .png()
    .toBuffer();
}

async function writeBothWebp(name: string, pngBuf: Buffer): Promise<void> {
  const webp = await sharp(pngBuf).webp({ quality: 88 }).toBuffer();
  fs.writeFileSync(path.join(OUT_DIR, name), webp);
  fs.writeFileSync(path.join(DOC_DIR, name), webp);
}

async function main(): Promise<void> {
  fs.mkdirSync(OUT_DIR, { recursive: true });
  fs.mkdirSync(DOC_DIR, { recursive: true });

  const keys = ["b13ref", "candB7", "candB13"] as const;
  for (const axis of ["fur", "hair", "metal"] as const) {
    const tiles = await Promise.all(
      keys.map((key) => labeledCrop(sources[key].file, sources[key][axis], `${sources[key].label} ${axis}`))
    );
    await writeBothWebp(`compare-native-${axis}.webp`, await row(tiles));
  }

  const thumbs = await Promise.all([
    labeledThumb(sources.candB7.file, sources.candB7.label),
    labeledThumb(sources.candB13.file, sources.candB13.label),
  ]);
  await writeBothWebp("compare-full-b7-b13-candidates.webp", await row(thumbs, 10));
  console.log("wrote comparison strips");
}

main().catch((error) => {
  console.error(error);
  process.exit(1);
});
