function sha256(buf: Buffer): string {
  return createHash("sha256").update(buf).digest("hex");
}

function writeEvidence(payload: Record<string, unknown>): void {
  fs.mkdirSync(path.dirname(EVIDENCE), { recursive: true });
  fs.writeFileSync(EVIDENCE, JSON.stringify(payload, null, 1));
  fs.mkdirSync(ARTIFACT_DIR, { recursive: true });
  fs.writeFileSync(path.join(ARTIFACT_DIR, EVIDENCE_BASENAME), JSON.stringify(payload, null, 1));
}

function resolveHttpsPublicUrl(stylePath: string, env: NodeJS.ProcessEnv): string {
  const rawBase =
    env.NEXTAUTH_URL?.trim() ||
    env.RAILWAY_STATIC_URL?.trim() ||
    (env.RAILWAY_PUBLIC_DOMAIN?.trim() ? `https://${env.RAILWAY_PUBLIC_DOMAIN.trim()}` : "");
  if (!rawBase) throw new Error("style ref requires NEXTAUTH_URL or Railway public URL");
  const base = rawBase.startsWith("http://") || rawBase.startsWith("https://") ? rawBase : `https://${rawBase}`;
  const url = new URL(stylePath, base);
  if (url.protocol !== "https:") throw new Error("style ref must be https");
  return url.toString();
}

/** Paid-call gate: deps + writable evidence paths before any provider HTTP. */
async function assertPaidRunPreflight(): Promise<void> {
  const probe = await sharp({
    create: { width: 2, height: 2, channels: 3, background: "#000" },
  })
    .webp()
    .toBuffer();
  if (!probe.length) throw new Error("sharp preflight produced empty webp");

  for (const dir of [ARTIFACT_DIR, DOC_DIR]) {
    fs.mkdirSync(dir, { recursive: true });
    const probePath = path.join(dir, `.sig4-write-probe-${RUN_LABEL}`);
    fs.writeFileSync(probePath, probe);
    fs.rmSync(probePath);
  }

  const rawArtifact = path.join(ARTIFACT_DIR, RAW_PRESERVE_BASENAME);
  const rawDoc = path.join(DOC_DIR, RAW_PRESERVE_BASENAME);
  fs.writeFileSync(rawArtifact, probe);
  fs.writeFileSync(rawDoc, probe);
  fs.rmSync(rawArtifact);
  fs.rmSync(rawDoc);
}

function preserveProviderBytes(buffer: Buffer): { artifact: string; doc: string; sha256: string } {
  fs.mkdirSync(ARTIFACT_DIR, { recursive: true });
  fs.mkdirSync(DOC_DIR, { recursive: true });
  const artifact = path.join(ARTIFACT_DIR, RAW_PRESERVE_BASENAME);
  const doc = path.join(DOC_DIR, RAW_PRESERVE_BASENAME);
  fs.writeFileSync(artifact, buffer);
  fs.writeFileSync(doc, buffer);
  return { artifact, doc, sha256: sha256(buffer) };
}

function readPreservedRaw(): Buffer | null {
  const artifact = path.join(ARTIFACT_DIR, RAW_PRESERVE_BASENAME);
  const doc = path.join(DOC_DIR, RAW_PRESERVE_BASENAME);
  if (fs.existsSync(artifact)) return fs.readFileSync(artifact);
  if (fs.existsSync(doc)) return fs.readFileSync(doc);
  return null;
}

        preservedRaw = preserveProviderBytes(result.buffer);
        transmitted.preservedRaw = preservedRaw;
        return {
          buffer: result.buffer,
          costUsd: result.costUsd,
          hasUnknownAttemptCost: result.costUsd == null,
          providerRequestId: result.providerRequestId,
        };
      } catch (error) {
        const message = error instanceof Error ? error.message : String(error);
        const diagnostic =
          error && typeof error === "object" && "diagnostic" in error
            ? (error as { diagnostic?: Record<string, unknown> }).diagnostic
            : null;
        writeEvidence({
          ...preflight,
          executed: true,
          failed: true,
          error: message,
          diagnostic,
          transmitted,
          providerPostsThisRun,
        });
        const cost =
          diagnostic && typeof diagnostic.computedCostUsd === "number" ? diagnostic.computedCostUsd : null;
        throw new OfficialImageTransportError(message, cost, cost == null, true);
      }
    },
  };

  const spool = new Map<string, Buffer>();
  const deps: OfficialAssetRunnerDeps = {
    store,
    transport: oneShot,
    imageOps: {
      async inspect(buffer) {
        const meta = await sharp(buffer, { failOn: "none" }).metadata();
        return meta.width && meta.height ? { width: meta.width, height: meta.height } : null;
      },
    },
    storage: {
      async store(filename, buffer) {
        const dest = path.join(uploadsDataDir(), filename);
        fs.writeFileSync(dest, buffer);
        return { url: `/uploads/${filename}` };
      },
    },
    spool: {
      async save(key, buffer) {
        spool.set(key, buffer);
      },
      async load(key) {
        return spool.get(key) ?? null;
      },
      async remove(key) {
        spool.delete(key);
      },
    },
    workerId: `sig4-oneshot-${RUN_LABEL}`,
    env,
  };

  let generated = await runOfficialAssetReplacement(deps, draftKey, SLOT);
  let replacement = store.getReplacement(draftKey, SLOT);
  let recoveredFromPreservedRaw = false;

  if (generated.status !== "generated") {
    const raw = readPreservedRaw();
    if (raw && raw.length > 0) {
      const meta = await sharp(raw, { failOn: "none" }).metadata();
      const dims =
        meta.width && meta.height ? { width: meta.width, height: meta.height } : null;
      if (dims && evaluateOfficialImageDimensions(composed.profile, dims.width, dims.height) === "exact") {
        const filename = `official-pilot-rf-v4-02__sig4-r${replacement.attempts}.webp`;
        const stored = await deps.storage.store(filename, raw);
        if (store.completeReplacement(draftKey, SLOT, deps.workerId, {
          url: stored.url,
          width: dims.width,
          height: dims.height,
        })) {
          recoveredFromPreservedRaw = true;
          generated = { status: "generated", url: stored.url, model: model };
          replacement = store.getReplacement(draftKey, SLOT);
        }
      }
    }
  }

  const imageCost = replacement.spentUsd;
  let vision: Awaited<ReturnType<typeof analyzeAssetImage>> | null = null;
  let visionError: string | null = null;
  const visionBudgetOk = imageCost + 0.001 < BUDGET_USD;
  if (generated.status === "generated" && replacement.candidateResultUrl && visionBudgetOk && routerKey) {
    try {
      vision = await analyzeAssetImage(replacement.candidateResultUrl);
    } catch (error) {
      visionError = error instanceof Error ? error.message : String(error);
    }
  }

  const candidatePath = replacement.candidateResultUrl
    ? path.join(uploadsDataDir(), path.basename(replacement.candidateResultUrl))
    : null;
  let candidateBuf =
    candidatePath && fs.existsSync(candidatePath) ? fs.readFileSync(candidatePath) : readPreservedRaw();
  if (candidateBuf) {
    fs.writeFileSync(path.join(ARTIFACT_DIR, CANDIDATE_BASENAME), candidateBuf);
    fs.writeFileSync(path.join(DOC_DIR, CANDIDATE_BASENAME), candidateBuf);
  }

  const runKnownUsd =
    imageCost + (vision?.costAttempts ?? []).reduce((sum, row) => sum + (row.upstreamCostUsd ?? 0), 0);

  const actual = {
    ...preflight,
    executed: true,
    generateStatus: generated.status,
    generateError: "error" in generated ? generated.error : null,
    recoveredFromPreservedRaw,
    providerPostsThisRun,
    providerRequestId: replacement.providerRequestId,
    imageCostUsd: imageCost,
    hasUnknownCost: replacement.hasUnknownCost,
    candidateUrl: replacement.candidateResultUrl,
    candidateSha256: candidateBuf ? sha256(candidateBuf) : null,
    candidateBytes: candidateBuf?.length ?? null,
    width: replacement.width,
    height: replacement.height,
    modelStored: replacement.model,
    transmitted,
    vision: vision
      ? {
          tag: vision.tag,
          estimated: vision.estimated,
          adultFlagged: vision.adultFlagged,
          moderationReject: vision.moderationReject,
          reason: vision.moderationReason,
          costAttempts: vision.costAttempts,
        }
      : null,
    visionError,
    visionSkipped: !vision && !visionError,
    runKnownUsd,
    cumulativeKnownUsdIncludingPriorCampaign: PRIOR_CAMPAIGN_TOTAL_USD + runKnownUsd,
    applied: false,
  };
  writeEvidence(actual);
  console.log(
    JSON.stringify(
      {
        styleCompare: STYLE_COMPARE,
        generateStatus: generated.status,
        providerPostsThisRun,
        providerRequestId: replacement.providerRequestId,
        imageCostUsd: imageCost,
        candidateSha256: actual.candidateSha256,
        visionTag: vision?.tag ?? null,
        runKnownUsd,
        cumulativeKnownUsdIncludingPriorCampaign: actual.cumulativeKnownUsdIncludingPriorCampaign,
        transmittedReferences: transmitted.referencesActuallySent,
        transmittedPreparedRefs: transmitted.preparedRefs,
      },
      null,
      1
    )
  );
}

main().catch((error) => {
  writeEvidence({
    executed: true,
    failed: true,
    error: error instanceof Error ? error.message : String(error),
    stack: error instanceof Error ? error.stack : null,
  });
  console.error(error);
  process.exit(1);
});
