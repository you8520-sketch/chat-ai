// Usage: node sig4-intake-snapshot.cjs <characterId> <out.json>
const Database = require("/workspace/node_modules/better-sqlite3");
const crypto = require("node:crypto");
const fs = require("node:fs");

const [characterId, out] = [Number(process.argv[2]), process.argv[3]];
const db = new Database("/tmp/sig4-intake-data/app.db", { readonly: true });
const sha = (v) => crypto.createHash("sha256").update(typeof v === "string" ? v : JSON.stringify(v)).digest("hex").slice(0, 16);

const character = db.prepare("SELECT * FROM characters WHERE id=?").get(characterId);
const columns = {};
for (const [key, value] of Object.entries(character)) {
  if (key === "_metadata") continue;
  columns[key] = value == null ? null : typeof value === "string" ? { len: value.length, sha: sha(value) } : value;
}
const assets = JSON.parse(character.assets);

const draft = db.prepare("SELECT draft_key FROM official_supply_characters WHERE staged_character_id=?").get(characterId);
const supply = {};
if (draft) {
  supply.character = db.prepare("SELECT * FROM official_supply_characters WHERE draft_key=?").get(draft.draft_key);
  supply.assets = db.prepare("SELECT * FROM official_supply_assets WHERE draft_key=? ORDER BY slot_key").all(draft.draft_key);
  supply.replacements = db.prepare("SELECT * FROM official_supply_asset_replacements WHERE draft_key=?").all(draft.draft_key);
}

const linked = {};
for (const { name } of db.prepare("SELECT name FROM sqlite_master WHERE type='table'").all()) {
  const cols = db.prepare(`PRAGMA table_info(${JSON.stringify(name)})`).all().map((c) => c.name);
  if (!cols.includes("character_id")) continue;
  const rows = JSON.parse(JSON.stringify(db.prepare(`SELECT * FROM ${JSON.stringify(name)} WHERE character_id=?`).all(characterId), (_k, v) => (_k === "_metadata" ? undefined : v)));
  linked[name] = { count: rows.length, sha: sha(rows) };
}
const strip = (_k, v) => (_k === "_metadata" ? undefined : v);
fs.writeFileSync(out, JSON.stringify({ columns, assets, supply, linked }, strip, 1));
console.log("snapshot", out, "assets", assets.length, "linkedTables", Object.keys(linked).length);
