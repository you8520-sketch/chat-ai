import fs from "node:fs";

async function main(): Promise<void> {
  if (process.env.DATA_DIR !== "/tmp/sig4-intake-data") throw new Error("isolated intake DATA_DIR required");
  const { getDb } = await import("@/lib/db");
  getDb();
  const { transferOfficialCharacterOwnerToAdmin } = await import("@/lib/officialSupply/ownerTransfer");
  const admin = { id: 1, email: "admin-intake@test.local", is_admin: 0 };
  const dry = transferOfficialCharacterOwnerToAdmin({ admin, characterId: 10, mode: "dry_run" });
  const apply = transferOfficialCharacterOwnerToAdmin({
    admin,
    characterId: 10,
    mode: "apply",
    preflightSnapshot: dry.preflightSnapshot,
  });
  const summary = {
    dry: { changedFields: dry.changedFields, fromOwnerId: dry.fromOwnerId, toOwnerId: dry.toOwnerId },
    apply: { applied: apply.applied, changedFields: apply.changedFields },
  };
  console.log(JSON.stringify(summary, null, 1));
  fs.writeFileSync("/tmp/intake/owner-transfer.json", JSON.stringify(summary, null, 1));
}

main().catch((error) => {
  console.error(error);
  process.exit(1);
});
