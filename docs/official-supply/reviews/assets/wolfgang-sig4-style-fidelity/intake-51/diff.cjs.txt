// Usage: node sig4-intake-diff.cjs <before.json> <after.json> <out.json>
const fs = require("node:fs");
const [beforePath, afterPath, outPath] = process.argv.slice(2);
const before = JSON.parse(fs.readFileSync(beforePath, "utf8"));
const after = JSON.parse(fs.readFileSync(afterPath, "utf8"));
const eq = (a, b) => JSON.stringify(a) === JSON.stringify(b);

const changedColumns = Object.keys(before.columns).filter((key) => !eq(before.columns[key], after.columns[key]));
const assetChanges = before.assets
  .map((asset, index) => (eq(asset, after.assets[index]) ? null : index))
  .filter((index) => index !== null);
const unchangedAssetIndexes = before.assets.map((_, i) => i).filter((i) => !assetChanges.includes(i));
const supplyChanged = {};
for (const key of ["character", "assets", "replacements"]) {
  supplyChanged[key] = !eq(before.supply[key], after.supply[key]);
}
const linkedChanged = Object.keys(before.linked).filter((table) => !eq(before.linked[table], after.linked[table]));

const preserved = {
  manuscript: ["system_prompt", "greeting", "world", "example_dialog", "appearance_raw", "appearance_compiled", "creator_raw_description"].map((key) => ({
    column: key,
    same: eq(before.columns[key], after.columns[key]),
  })),
  officialVisibilityNsfw: ["official", "visibility", "moderation_status", "nsfw", "creator_name", "creator_id", "adult_status", "participant_min_age"].map((key) => ({
    column: key,
    same: eq(before.columns[key], after.columns[key]),
    value: after.columns[key],
  })),
  representative: {
    beforeRep: before.assets.findIndex((a) => a.representativeRank != null),
    afterRep: after.assets.findIndex((a) => a.representativeRank != null),
    sameObject: eq(before.assets.find((a) => a.representativeRank != null), after.assets.find((a) => a.representativeRank != null)),
  },
  blurFlagsOutsideSig4: before.assets.every((a, i) => i === 4 || a.viewerBlur === after.assets[i].viewerBlur),
};

const result = {
  changedColumns,
  changedAssetIndexes: assetChanges,
  unchangedAssetCount: unchangedAssetIndexes.length,
  supplyChanged,
  linkedTablesChanged: linkedChanged,
  lorebookAttachments: { before: before.linked.character_lorebook_attachments, after: after.linked.character_lorebook_attachments },
  preserved,
  sig4Before: before.assets[4],
  sig4After: after.assets[4],
  supplySig4Row: {
    before: before.supply.assets.find((a) => a.slot_key === "sig4")?.result_url,
    after: after.supply.assets.find((a) => a.slot_key === "sig4")?.result_url,
  },
  driftBetweenPublicAndSupplySig4:
    after.assets[4].url !== after.supply.assets.find((a) => a.slot_key === "sig4")?.result_url,
};
fs.writeFileSync(outPath, JSON.stringify(result, null, 1));
console.log(JSON.stringify(result, null, 1));
