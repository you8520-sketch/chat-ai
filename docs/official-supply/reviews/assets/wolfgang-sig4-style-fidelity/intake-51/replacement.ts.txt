/**
 * Existing official replacement owner, fed with an existing external file. No provider, no vision.
 * Runs on a COPY of the isolated intake DB (DATA_DIR=/tmp/sig4-intake-copy).
 */
import fs from "node:fs";
import path from "node:path";
import { createRequire } from "node:module";

const sharp = createRequire("/workspace/package.json")("sharp") as typeof import("sharp");
const COPY_DIR = "/tmp/sig4-intake-copy";
const EXTERNAL_FILE =
  "/workspace/docs/official-supply/reviews/assets/wolfgang-sig4-style-fidelity/sig4-candidate-b13.webp";
const DRAFT = "pilot-rf-v4-02";
const SLOT = "sig4";

async function main(): Promise<void> {
  if (process.env.DATA_DIR !== COPY_DIR) throw new Error("DATA_DIR must be the intake copy");
  const { getDb } = await import("@/lib/db");
  const { OfficialSupplyStore, OfficialSupplyGateError } = await import("@/lib/officialSupply/store");
  const { applyApprovedOfficialAssetReplacements } = await import("@/lib/officialSupply/replacement");
  const { moderateOfficialAssetReplacement } = await import("@/lib/officialSupply/runner");
  const { uploadsDataDir } = await import("@/lib/uploadStorage");

  const db = getDb();
  const store = new OfficialSupplyStore(db);
  const result: Record<string, unknown> = {};

  const block = (fn: () => unknown): string => {
    try {
      fn();
      return "NOT_BLOCKED";
    } catch (error) {
      return error instanceof OfficialSupplyGateError ? `${error.code}: ${error.message}` : String(error);
    }
  };
  result.stagedPrivate = {
    stage: store.getCharacter(DRAFT).stage,
    createReplacementCandidate: block(() => store.createReplacementCandidate(DRAFT, SLOT)),
  };

  const characterId = store.getCharacter(DRAFT).stagedCharacterId!;
  db.prepare("UPDATE characters SET official=1, visibility='public', moderation_status='approved' WHERE id=?").run(characterId);
  store.markPublished(DRAFT, characterId);
  result.afterPublishInCopy = { stage: store.getCharacter(DRAFT).stage };

  const before = db.prepare("SELECT assets FROM characters WHERE id=?").get(characterId) as { assets: string };
  const beforeAssets = JSON.parse(before.assets) as Array<Record<string, unknown>>;

  const created = store.createReplacementCandidate(DRAFT, SLOT);
  const claimed = store.claimReplacement(DRAFT, SLOT, "intake-operator", Date.now(), 60_000, 3);
  const meta = await sharp(EXTERNAL_FILE).metadata();
  const filename = `official-${DRAFT}__${SLOT}-r${store.getReplacement(DRAFT, SLOT).attempts}.webp`;
  fs.copyFileSync(EXTERNAL_FILE, path.join(uploadsDataDir(), filename));
  const completed = store.completeReplacement(DRAFT, SLOT, "intake-operator", {
    url: `/uploads/${filename}`,
    width: meta.width!,
    height: meta.height!,
  });
  await moderateOfficialAssetReplacement(
    {
      store,
      moderator: {
        async moderate() {
          return { status: "checked", adultFlagged: false, moderationReject: false, reason: "intake stub (no provider)" };
        },
      },
    },
    DRAFT,
    SLOT
  );
  const PASS = { ok: true };
  const review = store.reviewReplacement(DRAFT, SLOT, {
    identityMatchesAnchor: PASS, expressionMatchesPlan: PASS, characterPresent: PASS, artStyle: PASS,
  });
  const applied = applyApprovedOfficialAssetReplacements({ store, draftKey: DRAFT, slotKeys: [SLOT] });

  const after = db.prepare("SELECT assets FROM characters WHERE id=?").get(characterId) as { assets: string };
  const afterAssets = JSON.parse(after.assets) as Array<Record<string, unknown>>;
  const changedIndexes = afterAssets
    .map((asset, index) => (JSON.stringify(asset) === JSON.stringify(beforeAssets[index]) ? -1 : index))
    .filter((index) => index >= 0);
  const replacement = store.getReplacement(DRAFT, SLOT);
  const activeRow = store.getAsset(DRAFT, SLOT);
  result.externalFilePublishedCopy = {
    created: created.status,
    claimed,
    completed,
    review,
    applied,
    changedAssetIndexes: changedIndexes,
    sig4After: afterAssets[4],
    replacementRow: {
      status: replacement.status,
      spentUsd: replacement.spentUsd,
      hasUnknownCost: replacement.hasUnknownCost,
      model: replacement.model,
      providerRequestId: replacement.providerRequestId,
    },
    activeSupplyRow: { status: activeRow.status, model: activeRow.model, providerRequestId: activeRow.providerRequestId, resultUrl: activeRow.resultUrl },
    tagUnchangedBySwap: afterAssets[4]?.tag === beforeAssets[4]?.tag,
    dimensionGate: `${meta.width}x${meta.height} required exactly 1536x1024 for signature slots`,
  };
  console.log(JSON.stringify(result, null, 1));
  fs.writeFileSync("/tmp/intake/replacement-owner.json", JSON.stringify(result, null, 1));
}

main().catch((error) => {
  console.error(error);
  process.exit(1);
});
