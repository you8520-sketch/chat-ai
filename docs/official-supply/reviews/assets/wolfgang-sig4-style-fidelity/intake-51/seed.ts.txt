/**
 * Isolated seed: a #51-like Wolfgang supply character (staged_private, official=0, private)
 * in a throwaway DATA_DIR. No provider call, no production DB.
 */
import { createRequire } from "node:module";
import fs from "node:fs";
import path from "node:path";

const requireFromWorkspace = createRequire("/workspace/package.json");
const ISOLATED_DIR = "/tmp/sig4-intake-data";
const ADMIN_EMAIL = "admin-intake@test.local";
const ADMIN_PASSWORD = "intake-pass-1234";
const IMG_DIR = "/tmp/wf51img";

async function main(): Promise<void> {
  if (process.env.DATA_DIR !== ISOLATED_DIR) throw new Error("DATA_DIR must be the isolated intake dir");
  for (const key of ["OPENAI_API_KEY", "OPENROUTER_API_KEY", "CHEAPER_INFERENCE_API_KEY"]) {
    if (process.env[key]?.trim()) throw new Error(`${key} must be empty for the intake seed`);
  }
  for (const key of ["TURSO_DATABASE_URL", "DATABASE_URL", "BLOB_READ_WRITE_TOKEN"]) {
    if (process.env[key]?.trim()) throw new Error(`${key} must be unset for the intake seed`);
  }
  process.env.NEXTAUTH_URL ||= "https://hav.chat";

  const { getDb } = await import("@/lib/db");
  const { hashPassword } = await import("@/lib/auth");
  const { createSiteManagedStudioAccount } = await import("@/lib/siteManagedAccounts");
  const { loadApprovedWolfgangV4Source, WOLFGANG_V4_LOCKED_SNAPSHOT_TARGET } = await import(
    "@/lib/officialSupply/lockedSnapshotReconcileOperator"
  );
  const { HWANG_VOCAB, testAppearance, testAssetPlan, testDraft, testStyleCandidate } = await import(
    "@/lib/officialSupply/officialSupply.fixtures"
  );
  const { officialImageProfileForSlot } = await import("@/lib/officialSupply/imageProfile");
  const { PILOT_STYLE_PROOF_CANDIDATE_ID } = await import("@/lib/officialSupply/pilotStyleProof");
  const { runOfficialAssetSlot } = await import("@/lib/officialSupply/runner");
  const { stageOfficialCharacterPrivately } = await import("@/lib/officialSupply/staging");
  const { OfficialSupplyStore } = await import("@/lib/officialSupply/store");
  const { ROFAN_CLUSTER_B_VISUAL_STYLE_DNA } = await import("@/lib/officialSupply/style");
  const { uploadsDataDir } = await import("@/lib/uploadStorage");
  const { buildClusterBRofanStyleSeed, PILOT_STYLE_PROOF_V4_BATCH_KEY, PILOT_STYLE_PROOF_V4_STYLE_KEY } =
    await import("@/lib/officialSupply/userOwnedRofanStyleRefs");
  const { ROFAN_V4_PRODUCTION_BATCH_CONFIG } = await import("@/lib/officialSupply/pilotProduction");
  type OfficialAssetRunnerDeps = import("@/lib/officialSupply/runner").OfficialAssetRunnerDeps;
  type OfficialAssetModeration = import("@/lib/officialSupply/types").OfficialAssetModeration;
  type OfficialAnchorQaReport = import("@/lib/officialSupply/types").OfficialAnchorQaReport;
  type OfficialVariationQaReport = import("@/lib/officialSupply/types").OfficialVariationQaReport;

  const { getDatabasePath } = await import("@/lib/dataDir");
  const dbPath = getDatabasePath();
  if (!dbPath.startsWith(ISOLATED_DIR)) throw new Error(`refusing non-isolated database ${dbPath}`);
  const db = getDb();

  db.prepare(
    "INSERT OR IGNORE INTO users (email, nickname, pw_hash, points, is_adult) VALUES (?,?,?,2000,1)"
  ).run(ADMIN_EMAIL, "intake-admin", hashPassword(ADMIN_PASSWORD));
  const adminId = (db.prepare("SELECT id FROM users WHERE email=?").get(ADMIN_EMAIL) as { id: number }).id;
  const studio = createSiteManagedStudioAccount({
    nickname: "로맨스 공식 스튜디오 (intake)",
    email: "studio-intake@site-managed.invalid",
  });
  const stagingUser = { id: studio.id, nickname: studio.nickname, is_adult: 1 };

  const PASS = { ok: true };
  const anchorQa = (): OfficialAnchorQaReport => ({
    gender: PASS, ageAppearance: PASS, face: PASS, hair: PASS, eyes: PASS, body: PASS,
    identifyingFeatures: PASS, artStyle: PASS, outfit: PASS, cardCrop: PASS,
  });
  const variationQa = (): OfficialVariationQaReport => ({
    identityMatchesAnchor: PASS, expressionMatchesPlan: PASS, characterPresent: PASS, artStyle: PASS,
  });

  const { nextDraft, nextPlan } = loadApprovedWolfgangV4Source();
  const sourceSheet = JSON.parse(
    fs.readFileSync("/workspace/src/lib/officialSupply/pilot/characters/pilot-rf-02.json", "utf8")
  ) as { appearance: Parameters<InstanceType<typeof OfficialSupplyStore>["lockAppearance"]>[1] };

  const store = new OfficialSupplyStore(db);
  const batchKey = PILOT_STYLE_PROOF_V4_BATCH_KEY;
  const styleKey = PILOT_STYLE_PROOF_V4_STYLE_KEY;
  const env = process.env;
  const styleSeed = buildClusterBRofanStyleSeed(env);
  store.createBatch(batchKey, {
    ...ROFAN_V4_PRODUCTION_BATCH_CONFIG,
    budgetUsd: { ...ROFAN_V4_PRODUCTION_BATCH_CONFIG.budgetUsd, batch: 100, perCharacter: 2.5 },
  });
  store.proposeStyle({
    styleKey,
    genre: "로맨스 판타지",
    candidates: [
      { ...testStyleCandidate("c1"), dna: ROFAN_CLUSTER_B_VISUAL_STYLE_DNA },
      { ...testStyleCandidate(PILOT_STYLE_PROOF_CANDIDATE_ID), dna: ROFAN_CLUSTER_B_VISUAL_STYLE_DNA },
      testStyleCandidate("c3"),
    ],
    proofAssetLimit: 3,
  });
  store.approveStyleCandidate(styleKey, PILOT_STYLE_PROOF_CANDIDATE_ID, styleSeed, "owner");

  const fakeDeps = (): OfficialAssetRunnerDeps => ({
    store,
    transport: {
      async generate(input) {
        const [w, h] = input.size.split("x").map(Number) as [number, number];
        return { buffer: Buffer.from(`${w}x${h}`), costUsd: 0, hasUnknownAttemptCost: false, providerRequestId: "fake-proof" };
      },
    },
    imageOps: {
      async inspect(buffer) {
        const match = /^(\d+)x(\d+)$/.exec(buffer.toString());
        return match ? { width: Number(match[1]), height: Number(match[2]) } : null;
      },
    },
    storage: { async store(filename) { return { url: `/uploads/${filename}` }; } },
    spool: { async save() {}, async load() { return null; }, async remove() {} },
    workerId: "intake-seed-proof",
    env,
  });

  const proofDraft = testDraft({
    draftKey: "intake-style-proof",
    name: "프루프",
    vocabulary: HWANG_VOCAB,
    styleKey,
    worldKey: nextDraft.worldKey,
  });
  store.addCharacterDraft(batchKey, proofDraft, { isStyleProof: true });
  if (!store.lockText(proofDraft.draftKey, { stagingUser }).ok) throw new Error("proof lockText");
  if (!store.lockAppearance(proofDraft.draftKey, testAppearance()).ok) throw new Error("proof lockAppearance");
  if (!store.lockAssetPlan(proofDraft.draftKey, testAssetPlan()).ok) throw new Error("proof lockAssetPlan");
  const proofRun = await runOfficialAssetSlot(fakeDeps(), proofDraft.draftKey, "rep");
  if (proofRun.status !== "generated") throw new Error("proof generate");
  store.decideStyleProof(styleKey, "approve", "owner");

  const draftKey = nextDraft.draftKey;
  store.addCharacterDraft(batchKey, { ...nextDraft, styleKey, worldKey: nextDraft.worldKey });
  const textQa = store.lockText(draftKey, { stagingUser });
  if (!textQa.ok) throw new Error(`wolfgang lockText ${JSON.stringify(textQa.errors)}`);
  if (!store.lockAppearance(draftKey, sourceSheet.appearance).ok) throw new Error("wolfgang lockAppearance");
  if (!store.lockAssetPlan(draftKey, nextPlan).ok) throw new Error("wolfgang lockAssetPlan");

  const character = store.getCharacter(draftKey);
  const clean: OfficialAssetModeration = { status: "checked", adultFlagged: false, moderationReject: false, reason: "" };
  const uploads = uploadsDataDir();
  fs.mkdirSync(uploads, { recursive: true });
  const available = fs.readdirSync(IMG_DIR).filter((name) => name.endsWith(".webp"));
  for (const [index, slot] of character.assetPlan!.slots.entries()) {
    const profile = officialImageProfileForSlot(slot.kind);
    const filename =
      slot.slotKey === "rep"
        ? path.basename(WOLFGANG_V4_LOCKED_SNAPSHOT_TARGET.expectedRepUrl)
        : `official-${draftKey}__${slot.slotKey}-a1.webp`;
    const url = `/uploads/${filename}`;
    const sourceName = fs.existsSync(path.join(IMG_DIR, `${slot.slotKey}.webp`))
      ? `${slot.slotKey}.webp`
      : available[index % available.length]!;
    fs.copyFileSync(path.join(IMG_DIR, sourceName), path.join(uploads, filename));
    store.database
      .prepare(
        `UPDATE official_supply_assets
         SET status='approved', result_url=?, width=?, height=?, attempts=1,
             appearance_lock_hash=?, moderation_json=?, qa_json=?
         WHERE draft_key=? AND slot_key=?`
      )
      .run(
        url, profile.width, profile.height, character.appearanceLockHash, JSON.stringify(clean),
        JSON.stringify(slot.kind === "representative" ? anchorQa() : variationQa()), draftKey, slot.slotKey
      );
  }
  store.database
    .prepare("UPDATE official_supply_characters SET stage='qa_passed', updated_at=datetime('now') WHERE draft_key=?")
    .run(draftKey);

  const staged = await stageOfficialCharacterPrivately({ store, draftKey, stagingUser });
  const row = db
    .prepare("SELECT id, creator_id, official, visibility, moderation_status FROM characters WHERE id=?")
    .get(staged.characterId);
  console.log(JSON.stringify({ seeded: true, draftKey, characterId: staged.characterId, adminId, studioId: studio.id, row }, null, 1));
}

main().catch((error) => {
  console.error(error);
  process.exit(1);
});
