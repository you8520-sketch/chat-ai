import { createHash } from "node:crypto";
import { createRequire } from "node:module";
import fs from "node:fs";
import http from "node:http";
import path from "node:path";

const { chromium } = createRequire("/workspace/package.json")("playwright");

const DOC = "/workspace/docs/official-supply/reviews/assets/wolfgang-asset-redesign";
const ART = "/opt/cursor/artifacts";
const REP = "/tmp/habby-test-db-X3VSSs/uploads/official-pilot-rf-v4-02__rep-a1.webp";
const F = path.join(DOC, "rep-f.webp");
const EXPECTED_REP = "a66c49d4d947209206b70a284c99f2e26ef868934ed5426818042c29ba3a45f1";

function sha256(file) {
  return createHash("sha256").update(fs.readFileSync(file)).digest("hex");
}

const repSha = sha256(REP);
if (repSha !== EXPECTED_REP) throw new Error(`rep sha ${repSha}`);
fs.copyFileSync(REP, path.join(DOC, "production-rep.webp"));
fs.copyFileSync(REP, path.join(ART, "production-rep.webp"));

const server = http.createServer((req, res) => {
  const file = req.url === "/rep.webp" ? REP : req.url === "/f.webp" ? F : "";
  if (!file) {
    res.writeHead(404);
    res.end();
    return;
  }
  res.writeHead(200, { "content-type": "image/webp" });
  res.end(fs.readFileSync(file));
});
await new Promise((resolve) => server.listen(0, "127.0.0.1", resolve));
const port = server.address().port;
