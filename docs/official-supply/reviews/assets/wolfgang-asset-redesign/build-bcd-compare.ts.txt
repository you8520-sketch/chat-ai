import { createHash } from "node:crypto";
import { createRequire } from "node:module";
import fs from "node:fs";
import path from "node:path";

const requireFromWorkspace = createRequire("/workspace/package.json");
const sharp = requireFromWorkspace("sharp") as typeof import("sharp");

const DOC = "/workspace/docs/official-supply/reviews/assets/wolfgang-asset-redesign";
const ART = "/opt/cursor/artifacts";
const sources = {
  B: path.join(DOC, "rep-b-raw.webp"),
  C: path.join(DOC, "rep-c-raw.webp"),
  D: path.join(DOC, "rep-d-raw.webp"),
} as const;

const nativeWindows = {
  face: { left: 282, top: 180, width: 460, height: 460, scale: "1:1 extract, no resize" },
  hair: { left: 312, top: 0, width: 400, height: 400, scale: "1:1 extract, no resize" },
  fur: {
    left: 80,
    top: 560,
    width: 220,
    height: 180,
    scale: "1:1 extract, no resize",
    note: "fur mass only. Excludes metal, skin, and background.",
  },
  cuirass: {
    left: 390,
    top: 700,
    width: 280,
    height: 200,
    scale: "1:1 extract, no resize",
    note: "chest-plate plane. Excludes glove and skin.",
  },
  jawCharacterRight: {
    left: 282,
    top: 430,
    width: 230,
    height: 210,
    scale: "1:1 extract, no resize",
    note: "viewer-left window. Character's right jaw on a frontal face.",
  },
  jawCharacterLeft: {
    left: 512,
    top: 430,
    width: 230,
    height: 210,
    scale: "1:1 extract, no resize",
    note: "viewer-right window. Character's left jaw on a frontal face.",
  },
} as const;

function sha256(buf: Buffer): string {
  return createHash("sha256").update(buf).digest("hex");
}

async function extract(file: string, box: { left: number; top: number; width: number; height: number }) {
  const meta = await sharp(file).metadata();
  if (meta.width !== 1024 || meta.height !== 1536) {
    throw new Error(`${file} is ${meta.width}x${meta.height}`);
  }
  return sharp(file).extract(box).webp({ quality: 92 }).toBuffer();
}

async function labeledRow(items: Array<{ label: string; buf: Buffer }>, cellW: number, cellH: number) {
  const labeled = await Promise.all(
    items.map(async (item) => {
      const bar = await sharp({
        create: { width: cellW, height: 36, channels: 3, background: "#111111" },
      })
        .composite([
          {
            input: Buffer.from(
              `<svg width="${cellW}" height="36"><text x="8" y="24" font-size="16" fill="#ffffff" font-family="sans-serif">${item.label}</text></svg>`
            ),
            top: 0,
            left: 0,
          },
        ])
        .png()
        .toBuffer();
      return sharp({
        create: { width: cellW, height: cellH + 36, channels: 3, background: "#000000" },
      })
        .composite([
          { input: bar, top: 0, left: 0 },
          { input: item.buf, top: 36, left: 0 },
        ])
        .png()
        .toBuffer();
    })
  );
  return sharp({
    create: { width: cellW * labeled.length, height: cellH + 36, channels: 3, background: "#000000" },
  })
    .composite(labeled.map((buf, index) => ({ input: buf, top: 0, left: index * cellW })))
    .webp({ quality: 90 })
    .toBuffer();
}

async function main(): Promise<void> {
  const outputs: Record<string, unknown> = {};
  const fullCells = await Promise.all(
    (["B", "C", "D"] as const).map(async (label) => ({
      label: `${label} 0.5x`,
      buf: await sharp(sources[label]).resize(512, 768, { fit: "fill" }).webp({ quality: 90 }).toBuffer(),
    }))
  );
  const full = await labeledRow(fullCells, 512, 768);
  const fullName = "compare-bcd-full.webp";
  fs.writeFileSync(path.join(DOC, fullName), full);
  fs.writeFileSync(path.join(ART, fullName), full);
  outputs[fullName] = { sha256: sha256(full), bytes: full.length, scale: "each panel 0.5 of native 1024x1536, same scale across B/C/D" };

  for (const name of ["face", "hair", "fur", "cuirass"] as const) {
    const box = nativeWindows[name];
    const cells = await Promise.all(
      (["B", "C", "D"] as const).map(async (label) => ({
        label: `${label} 1:1`,
        buf: await extract(sources[label], box),
      }))
    );
    const row = await labeledRow(cells, box.width, box.height);
    const fileName = `compare-bcd-${name}.webp`;
    fs.writeFileSync(path.join(DOC, fileName), row);
    fs.writeFileSync(path.join(ART, fileName), row);
    outputs[fileName] = { sha256: sha256(row), bytes: row.length, window: box };
  }

  const right = nativeWindows.jawCharacterRight;
  const left = nativeWindows.jawCharacterLeft;
  const top = await labeledRow(
    await Promise.all(
      (["B", "C", "D"] as const).map(async (label) => ({
        label: `${label} char-right`,
        buf: await extract(sources[label], right),
      }))
    ),
    right.width,
    right.height
  );
  const bottom = await labeledRow(
    await Promise.all(
      (["B", "C", "D"] as const).map(async (label) => ({
        label: `${label} char-left`,
        buf: await extract(sources[label], left),
      }))
    ),
    left.width,
    left.height
  );
  const jaw = await sharp({
    create: { width: right.width * 3, height: (right.height + 36) * 2, channels: 3, background: "#000000" },
  })
    .composite([
      { input: top, top: 0, left: 0 },
      { input: bottom, top: right.height + 36, left: 0 },
    ])
    .webp({ quality: 92 })
    .toBuffer();
  fs.writeFileSync(path.join(DOC, "compare-bcd-jaw.webp"), jaw);
  fs.writeFileSync(path.join(ART, "compare-bcd-jaw.webp"), jaw);
  outputs["compare-bcd-jaw.webp"] = {
    sha256: sha256(jaw),
    bytes: jaw.length,
    scale: "1:1 extract, no resize",
    topRow: right,
    bottomRow: left,
  };

  const record = {
    keptFixedSheets: [
      "compare-abc-full.webp",
      "compare-abc-face.webp",
      "compare-abc-hair.webp",
      "compare-abc-fur.webp",
      "compare-abc-metal.webp",
    ],
    sources: Object.fromEntries(
      (["B", "C", "D"] as const).map((label) => {
        const bytes = fs.readFileSync(sources[label]);
        return [label, { file: path.basename(sources[label]), sha256: sha256(bytes), bytes: bytes.length }];
      })
    ),
    outputs,
  };
  const text = `${JSON.stringify(record, null, 2)}\n`;
  fs.writeFileSync(path.join(DOC, "compare-bcd-windows.json"), text);
  fs.writeFileSync(path.join(ART, "compare-bcd-windows.json"), text);
  console.log(JSON.stringify(Object.keys(outputs)));
}

main().catch((error) => {
  console.error(error instanceof Error ? error.message : String(error));
  process.exit(1);
});
