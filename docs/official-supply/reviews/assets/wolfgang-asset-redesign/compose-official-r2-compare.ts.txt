function gateFields(draft: OfficialCharacterDraft) {
  return {
    draftKey: draft.draftKey,
    name: draft.name,
    gender: draft.gender,
    age: draft.age,
    genres: draft.genres,
    audience: draft.audience,
    matureMaleTarget: isMatureMaleRofanRomanceTarget(draft),
    matureMalePromptPresent: Boolean(buildMatureMaleVisualAgePrompt(draft)),
  };
}

function sha256(text: string): string {
  return createHash("sha256").update(text, "utf8").digest("hex");
}

function wolfgangCharacter(
  draft: OfficialCharacterDraft,
  appearance: OfficialAppearanceLock,
  plan: OfficialAssetPlan
): OfficialCharacterRecord {
  return {
    draftKey: "pilot-rf-02",
    batchKey: "pilot-romance-fantasy-04-cluster-b",
    worldKey: draft.worldKey,
    styleKey: draft.styleKey,
    stage: "asset_plan_locked",
    draft,
    textLockHash: null,
    appearance,
    appearanceLockHash: null,
    assetPlan: plan,
    isStyleProof: false,
    stagedCharacterId: null,
  };
}
