import Module from "node:module";

let sessionToken = "";
const originalLoad = Module._load;
Module._load = function (request, parent, isMain) {
  if (request === "server-only") return {};
  if (request === "next/headers") {
    return {
      cookies: async () => ({
        get: (name: string) =>
          name === "session" && sessionToken ? { value: sessionToken } : undefined,
      }),
    };
  }
  return originalLoad.call(this, request, parent, isMain);
} as typeof Module._load;

const dataDir = process.argv[2];
const model = process.argv[3];
const message = process.argv[4];
const outDir = process.argv[5];
const mode = process.argv[6];
if (!dataDir || !model || !message || !outDir || (mode !== "dry" && mode !== "live")) {
  throw new Error("usage: one-turn.mts <dataDir> <model> <message> <outDir> <dry|live>");
}

process.env.DATA_DIR = dataDir;
process.env.SESSION_SECRET =
  process.env.SESSION_SECRET || "local-b9-memory-recheck-secret-0123456789abcdef";
process.env.EPISODIC_MEMORY_RECALL_ENABLED = "1";
process.env.MEMORY_5PLUS4_ENABLED = "true";
process.env.CANON_INJECTION_ENABLED = "1";
process.env.CANON_INJECTION_ROLLOUT_STAGE = "D2";
process.env.EPISODIC_SEMANTIC_MODEL = "bge_m3";
process.env.BACKGROUND_MEMORY_MODEL = "gpt-6-luna";
delete process.env.EPISODIC_SEMANTIC_DISCOVERY_ENABLED;
delete process.env.MOCK_MODE;
delete process.env.PROMPT_DEBUG;

const { appendFileSync, mkdirSync, writeFileSync } = await import("node:fs");
const { resolveMainRpPrimaryWireModelId } = await import("@/lib/openRouterConfig");
const { getDb } = await import("@/lib/db");
const { createSession } = await import("@/lib/auth");

const expectedWire = resolveMainRpPrimaryWireModelId(model);
const db = getDb();
db.prepare("UPDATE users SET selected_ai=? WHERE id=1").run(model);
db.prepare("UPDATE chats SET gemini_model=? WHERE id=1").run(model);
sessionToken = createSession(1);

const calls: Array<{ url: string; model: string; bytes: number }> = [];
const originalFetch = globalThis.fetch;
globalThis.fetch = async (input: RequestInfo | URL, init?: RequestInit) => {
  const url = String(input instanceof Request ? input.url : input);
  if (/embedding|embeddings/i.test(url)) {
    throw new Error(`blocked non-main provider call: ${url.slice(0, 80)}`);
  }
  const isCompletion = url.includes("/chat/completions");
  let wireModel = "";
  let bodyText = "";
  if (typeof init?.body === "string") {
    bodyText = init.body;
    try {
      wireModel = String((JSON.parse(bodyText) as { model?: unknown }).model ?? "");
    } catch {
      wireModel = "";
    }
  }
  if (isCompletion) {
    calls.push({ url, model: wireModel, bytes: bodyText.length });
    mkdirSync(outDir, { recursive: true });
    writeFileSync(`${outDir}/request-${calls.length}.json`, bodyText);
    const mainCalls = calls.filter((call) => call.model === expectedWire).length;
    const blocked =
      mode === "dry" ||
      wireModel !== expectedWire ||
      mainCalls > 1 ||
      /embedding|embeddings/i.test(url);
    if (!blocked) {
      appendFileSync(
        "/tmp/b9/paid-calls.log",
        `${new Date().toISOString()} ${wireModel} ${url.slice(0, 80)}\n`
      );
    }
    if (blocked) {
      return new Response(
        `data: ${JSON.stringify({
          choices: [{ delta: { content: "dry" }, finish_reason: "stop" }],
          usage: { prompt_tokens: 1, completion_tokens: 1, cost: 0 },
        })}\n\ndata: [DONE]\n\n`,
        { status: 200, headers: { "Content-Type": "text/event-stream" } }
      );
    }
  }
  return originalFetch(input, init);
};

const { POST } = await import("@/app/api/chat/route");
const response = await POST(
  new Request("http://127.0.0.1/api/chat", {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({ characterId: 18, chatId: 1, message }),
  })
);
const raw = await response.text();
mkdirSync(outDir, { recursive: true });
writeFileSync(`${outDir}/sse.txt`, raw);
const events: Array<Record<string, unknown>> = [];
for (const line of raw.split("\n")) {
  if (!line.startsWith("data:")) continue;
  const payload = line.slice(5).trim();
  if (!payload || payload === "[DONE]") continue;
  try {
    events.push(JSON.parse(payload) as Record<string, unknown>);
  } catch {
    // non-json stream line
  }
}
const done = [...events].reverse().find((event) => event.type === "done");
const error = events.find((event) => event.type === "error");
const finalContent = typeof done?.finalContent === "string" ? done.finalContent : "";
writeFileSync(`${outDir}/final.txt`, finalContent);
writeFileSync(
  `${outDir}/meta.json`,
  JSON.stringify(
    {
      status: response.status,
      mode,
      selectedModel: model,
      expectedWire,
      calls,
      eventTypes: events.map((event) => event.type),
      error: error?.error ?? null,
      finalChars: finalContent.length,
      usage: done?.usage ?? null,
    },
    null,
    2
  )
);

    eventTypes: [...new Set(events.map((event) => String(event.type)))],
    error: error?.error ?? null,
    finalChars: finalContent.length,
  })
);
