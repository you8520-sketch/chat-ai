const OMIT = "…(오래된 기억 일부 생략)";
function textInfo(text: string | null | undefined) {
  const t = text ?? "";
  const first = /\[(\d+)~(\d+)턴\]/u.exec(t);
  return {
    len: t.length,
    sha12: sha(t),
    omitPrefix: t.includes(OMIT),
    firstBlock: first ? `${first[1]}~${first[2]}` : null,
  };
}
function snap(): Record<string, unknown> {
  if (!stateProbe || probing) return {};
  probing = true;
  try {
    return stateProbe();
  } catch (error) {
    return { probeError: String(error) };
  } finally {
    probing = false;
  }
}
function shortStack(): string[] {
  return (new Error().stack ?? "")
    .split("\n")
    .slice(2)
    .filter((line) => /\/src\/(lib|services|app)\//.test(line))
    .map((line) => line.trim().replace(/^at /, "").replace(/\(?\/workspace\//, "("))
    .slice(0, 8);
}
function log(event: Record<string, unknown>) {

}

Module._load = function (request: string, parent: unknown, isMain: boolean) {
  if (request === "server-only") return {};
  if (request === "next/headers") {
    return {
      cookies: async () => ({
        get: (name: string) =>
          name === "session" && sessionToken ? { value: sessionToken } : undefined,
      }),
    };
  }
  const exports = originalLoad.call(this, request, parent, isMain) as object;
  if (!exports || typeof exports !== "object") return exports;
  let filename = "";
  try {
    filename = (Module as unknown as { _resolveFilename: AnyFn })._resolveFilename(request, parent, isMain) as string;
  } catch {
    return exports;
  }
  const suffix = Object.keys(WRAPS).find((s) => filename.endsWith(s));
  if (!suffix) return exports;
  let proxy = proxies.get(exports);
  if (!proxy) {
    const wraps = WRAPS[suffix]!;
    proxy = new Proxy(exports, {
      get(target, prop, receiver) {
        const value = Reflect.get(target, prop, receiver);
        if (typeof prop === "string" && typeof value === "function" && wraps[prop]) {
          return wraps[prop]!(value as AnyFn);
        }
        return value;
      },
    });
    proxies.set(exports, proxy);
  }
  return proxy;
} as typeof Module._load;

function stage(name: string, extra?: Record<string, unknown>) {
  log({ kind: "stage", stage: name, ...extra, state: snap() });
}

function wrapCall(name: string, summarize?: (args: unknown[], result: unknown) => Record<string, unknown>) {
  return (fn: AnyFn): AnyFn =>
    function (this: unknown, ...args: unknown[]) {
      const before = snap();
      log({ kind: "enter", fn: name, state: before });
      const finish = (result: unknown) => {
        log({
          kind: "exit",
          fn: name,
          ...(summarize ? summarize(args, result) : {}),
          state: snap(),
        });
        return result;
      };
      let result: unknown;
      try {
        result = fn.apply(this, args);
      } catch (error) {
        log({ kind: "throw", fn: name, error: String(error) });
        throw error;
      }
      if (result && typeof (result as Promise<unknown>).then === "function") {
        return (result as Promise<unknown>).then(finish, (error) => {
          log({ kind: "reject", fn: name, error: String(error) });
          throw error;
        });
      }
      return finish(result);
    };
}

const Database = requireFromHere("better-sqlite3") as { prototype: { prepare: AnyFn; exec: AnyFn } };
const originalPrepare = Database.prototype.prepare;
Database.prototype.prepare = function (this: unknown, sql: string) {
  const stmt = originalPrepare.call(this, sql) as { run: AnyFn };
  if (/^\s*(UPDATE|INSERT|REPLACE|DELETE)\b[^;]*\bchat_memories\b/i.test(sql)) {
    const originalRun = stmt.run;
    stmt.run = function (this: unknown, ...args: unknown[]) {
      const before = snap();
      const result = originalRun.apply(this, args);
      log({
        kind: "chat_memories-write",
        sql: sql.replace(/\s+/g, " ").slice(0, 160),
        stack: shortStack(),
        before,
        state: snap(),
      });
      return result;
    };
  }
  return stmt;
};

stateProbe = () => {
  const row = db
    .prepare(
      `SELECT recent_summary, global_projection_kind, global_source_fingerprint, global_covered_through_turn,
              summarized_turn_count, message_count FROM chat_memories WHERE chat_id=1`
    )
    .get() as
    | {
        recent_summary: string;
        global_projection_kind: string | null;
        global_source_fingerprint: string | null;
        global_covered_through_turn: number | null;
        summarized_turn_count: number;
        message_count: number;
      }
    | undefined;
  if (!row) return { row: null };
  const covered = row.global_covered_through_turn;
  const computed = covered && covered > 0 ? buildPrefixFingerprintThroughTurn(1, covered) : null;
  const stored = row.global_source_fingerprint?.trim() || null;
  return {
    recent_summary: textInfo(row.recent_summary),
    kind: row.global_projection_kind,
    covered,
    fingerprintStored: stored ? stored.slice(0, 12) : null,
    fingerprintComputed: computed ? computed.slice(0, 12) : null,
    fingerprintMatches: stored != null && computed != null ? stored === computed : null,
    summarized_turn_count: row.summarized_turn_count,
    message_count: row.message_count,
  };
};

globalThis.fetch = async (input: RequestInfo | URL, init?: RequestInit) => {
  const url = String(input instanceof Request ? input.url : input);
  if (/embedding|embeddings/i.test(url)) {
    log({ kind: "blocked-embedding", url: url.slice(0, 80) });
    throw new Error(`blocked non-main provider call: ${url.slice(0, 80)}`);
  }
  let bodyText = "";
  let wireModel = "";
  if (typeof init?.body === "string") {
    bodyText = init.body;
    try {
      wireModel = String((JSON.parse(bodyText) as { model?: unknown }).model ?? "");
    } catch {
      wireModel = "";
    }
  }
  if (url.includes("/chat/completions")) {
    calls.push({ model: wireModel, bytes: bodyText.length, url: url.slice(0, 60) });
    const isMain = wireModel === expectedWire;
    const idx = bodyText.indexOf("[현재기억]");
    const block = idx >= 0 ? bodyText.slice(idx) : "";
    const decoded = (() => {
      try {
        return JSON.stringify(JSON.parse(bodyText));
      } catch {
        return bodyText;
      }
    })();
    log({
      kind: "provider-send-blocked",
      wireModel,
      isMain,
      bytes: bodyText.length,
      hasCurrentMemoryHeader: idx >= 0,
      omitPrefixInRequest: bodyText.includes("오래된 기억 일부 생략") || decoded.includes("오래된 기억 일부 생략"),
      firstBlockAfterHeader: /\[(\d+)~(\d+)턴\]/u.exec(block)?.slice(1, 3).join("~") ?? null,
      state: snap(),
    });
    if (isMain) {
      mkdirSync(outDir, { recursive: true });
      writeFileSync(`${outDir}/main-request.json`, bodyText);
    }
    return new Response(
      `data: ${JSON.stringify({
        choices: [{ delta: { content: "dry" }, finish_reason: "stop" }],
        usage: { prompt_tokens: 1, completion_tokens: 1, cost: 0 },
      })}\n\ndata: [DONE]\n\n`,
      { status: 200, headers: { "Content-Type": "text/event-stream" } }
    );
  }
  blockedOtherFetches += 1;
  log({ kind: "blocked-other-fetch", url: url.slice(0, 100) });
  throw new Error(`blocked other fetch: ${url.slice(0, 80)}`);
};
