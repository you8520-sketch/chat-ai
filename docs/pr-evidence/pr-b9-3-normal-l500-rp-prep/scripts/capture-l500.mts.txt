const originalLoad = Module._load;
Module._load = function (request: string, parent: unknown, isMain: boolean) {
  if (request === "server-only") return {};
  if (request === "next/headers") {
    return {
      cookies: async () => ({
        get: (name: string) =>
          name === "session" && sessionToken ? { value: sessionToken } : undefined,
      }),
    };
  }
  return originalLoad.call(this, request, parent, isMain);
} as typeof Module._load;

function rowSnapshot(): Record<string, unknown> {
  const db = getDb();
  const row = db
    .prepare(
      `SELECT recent_summary, global_projection_kind, global_source_fingerprint, global_covered_through_turn,
              summarized_turn_count, message_count FROM chat_memories WHERE chat_id=1`
    )
    .get() as
    | {
        recent_summary: string;
        global_projection_kind: string | null;
        global_source_fingerprint: string | null;
        global_covered_through_turn: number | null;
        summarized_turn_count: number;
        message_count: number;
      }
    | undefined;
  if (!row) return { missing: true };
  const covered = row.global_covered_through_turn;
  const computed =
    covered != null && covered > 0 ? buildPrefixFingerprintThroughTurn(1, covered) : null;
  const stored = row.global_source_fingerprint?.trim() || null;
  const checkpoint = readGlobalCheckpointSnapshot(row);
  return {
    recentLen: row.recent_summary.length,
    recentSha12: sha12(row.recent_summary),
    omitInStored: row.recent_summary.includes(OMIT),
    kind: row.global_projection_kind,
    covered,
    fingerprintMatches: stored != null && computed != null ? stored === computed : null,
    checkpointHealthy: isHealthyDurableGlobalCompactCheckpoint(1, checkpoint),
    summarized_turn_count: row.summarized_turn_count,
    message_count: row.message_count,
  };
}

const originalPrepare = Database.prototype.prepare;
Database.prototype.prepare = function (this: unknown, sql: string) {
  const stmt = originalPrepare.call(this, sql);
  if (/^\s*(UPDATE|INSERT|REPLACE|DELETE)\b[^;]*\bchat_memories\b/i.test(sql)) {
    const originalRun = stmt.run;
    stmt.run = function (this: unknown, ...args: unknown[]) {
      const before = rowSnapshot();
      const result = originalRun.apply(this, args);
      chatMemoryWrites.push({ sql: sql.replace(/\s+/g, " ").slice(0, 180), before, after: rowSnapshot() });
      return result;
    };
  }
  return stmt;
};

globalThis.fetch = async (input: RequestInfo | URL, init?: RequestInit) => {
  const url = String(input instanceof Request ? input.url : input);
  if (/embedding|embeddings/i.test(url)) {
    throw new Error(`blocked embedding: ${url.slice(0, 80)}`);
  }
  let bodyText = "";
  let wireModel = "";
  if (typeof init?.body === "string") {
    bodyText = init.body;
    try {
      wireModel = String((JSON.parse(bodyText) as { model?: unknown }).model ?? "");
    } catch {
      wireModel = "";
    }
  }
  if (url.includes("/chat/completions")) {
    const isMain = wireModel === expectedWire;
    completionCalls.push({ wireModel, bytes: bodyText.length, isMain });
    if (isMain) mainRequestBody = bodyText;
    return new Response(
      `data: ${JSON.stringify({
        choices: [{ delta: { content: "dry" }, finish_reason: "stop" }],
        usage: { prompt_tokens: 1, completion_tokens: 1, cost: 0 },
      })}\n\ndata: [DONE]\n\n`,
      { status: 200, headers: { "Content-Type": "text/event-stream" } }
    );
  }
  throw new Error(`blocked fetch: ${url.slice(0, 100)}`);
};

function extractBlock(full: string, header: string, nextHeaders: string[]): string {
  const start = full.indexOf(header);
  if (start < 0) return "";
  let end = full.length;
  for (const nh of nextHeaders) {
    const p = full.indexOf(nh, start + header.length);
    if (p >= 0) end = Math.min(end, p);
  }
  return full.slice(start, end).trim();
}

function parseMainRequest(body: string) {
  const parsed = JSON.parse(body) as {
    model?: string;
    messages?: Array<{ role?: string; content?: unknown }>;
    temperature?: number;
    top_p?: number;
    reasoning?: unknown;
    reasoning_effort?: unknown;
    provider?: unknown;
  };
  const messages = parsed.messages ?? [];
  const systemParts = messages
    .filter((m) => m.role === "system")
    .map((m) => (typeof m.content === "string" ? m.content : JSON.stringify(m.content)));
  const systemJoined = systemParts.join("\n\n---SYSTEM_PART---\n\n");
  const userParts = messages.filter((m) => m.role === "user").map((m) => String(m.content ?? ""));
  const currentUser = userParts[userParts.length - 1] ?? "";
  const memoryHeader = "[Memory]";
  const memStart = systemJoined.indexOf(memoryHeader);
  const memoryRegion = memStart >= 0 ? systemJoined.slice(memStart) : systemJoined;
  const currentMemory = extractBlock(memoryRegion, "[현재기억]", ["[중기", "[에피소드", "[관계", "[과거"]);
  const medium = extractBlock(memoryRegion, "[중기", ["[에피소드", "[관계", "[과거", "[Memory]"]);
  const episodic = extractBlock(memoryRegion, "[에피소드", ["[관계", "[과거", "[Memory]", "[현재"]);
  const relationship = extractBlock(memoryRegion, "[관계", ["[과거", "[Memory]", "[현재", "[에피소드"]);
  return {
    wireModel: parsed.model ?? "",
    messageCount: messages.length,
    systemPartCount: systemParts.length,
    systemPartHashes: systemParts.map((p) => sha12(p)),
    systemPartLens: systemParts.map((p) => p.length),
    systemJoinedLen: systemJoined.length,
    systemJoinedSha256: sha256(systemJoined),
    temperature: parsed.temperature ?? null,
    top_p: parsed.top_p ?? null,
    reasoning: parsed.reasoning ?? null,
    reasoning_effort: parsed.reasoning_effort ?? null,
    provider: parsed.provider ?? null,
    currentUserTurn: currentUser,
    memory: {
      currentMemory: {
        len: currentMemory.length,
        sha256: sha256(currentMemory),
        omitPrefix: currentMemory.includes(OMIT) || memoryRegion.includes(OMIT),
        body: currentMemory.replace(/^\[현재기억\][^\n]*\n[^\n]*\n\n?/u, "").trim(),
      },
      mediumTerm: {
        len: medium.length,
        sha256: sha256(medium),
        body: medium,
      },
      episodic: {
        len: episodic.length,
        sha256: sha256(episodic),
        body: episodic,
      },
      relationship: {
        len: relationship.length,
        sha256: sha256(relationship),
        body: relationship,
      },
    },
    requestBytes: body.length,
    max_tokens: (parsed as { max_tokens?: unknown }).max_tokens ?? null,
  };
}
