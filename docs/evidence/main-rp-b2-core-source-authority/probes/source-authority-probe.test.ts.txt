const originalLoad = (Module as unknown as { _load: typeof Module._load })._load;
(Module as unknown as { _load: typeof Module._load })._load = function (
  request: string,
  parent: NodeModule,
  isMain: boolean
) {
  if (request === "server-only") return {};
  return originalLoad(request, parent, isMain);
} as typeof Module._load;

type Ids = { userId: number; assistantId: number; userText: string; assistantText: string };
function turnIds(turn: number): Ids {
  const db = getDb();
  const u = db
    .prepare(`SELECT id, content FROM messages WHERE chat_id=? AND role='user' ORDER BY id LIMIT 1 OFFSET ?`)
    .get(CHAT, turn - 1) as { id: number; content: string };
  const a = db
    .prepare(`SELECT id, content FROM messages WHERE chat_id=? AND role='assistant' AND user_message_id=?`)
    .get(CHAT, u.id) as { id: number; content: string };
  return { userId: u.id, assistantId: a.id, userText: u.content, assistantText: a.content };
}

type Proposal = {
  label: string;
  turn: number;
  fact: Record<string, unknown>;
  evidence: { source_role: "user" | "assistant"; source_quote: string };
};

function mkFact(p: Proposal) {
  return {
    category: "character",
    subject: "x",
    attribute: "a",
    value: "v",
    importance: "important",
    ...p.fact,
    evidence: p.evidence,
  };
}

function runProposal(p: Proposal) {
  const ids = turnIds(p.turn);
  const db = getDb();
  const parsedSection = parseSharedEpisodicSection({ extracted_facts: [mkFact(p)] });
  const parsedFact = parsedSection.facts?.[0];
  const verdict = parsedFact
    ? evaluateEpisodicFactEvidence(
        parsedFact,
        loadEpisodicEvidenceSources(db, CHAT, { userMessageId: ids.userId, assistantMessageId: ids.assistantId }),
        { userPersona: PERSONA }
      )
    : null;
  const before = (db.prepare("SELECT COUNT(*) c FROM episodic_memory_facts WHERE chat_id=?").get(CHAT) as { c: number }).c;
  const result = reconcileSharedEpisodicFactsForTurn(db, {
    chatId: CHAT,
    userId: USER,
    characterId: CHAR,
    assistantMessageId: ids.assistantId,
    sourceUserMessageId: ids.userId,
    sourceUserText: ids.userText,
    userPersona: PERSONA,
    episodic: parseSharedEpisodicSection({ extracted_facts: [mkFact(p)] }),
    isRegeneration: false,
  });
  const row = db
    .prepare(
      "SELECT fact_text, source_user_message_id, metadata FROM episodic_memory_facts WHERE chat_id=? ORDER BY id DESC LIMIT 1"
    )
    .get(CHAT) as { fact_text: string; source_user_message_id: number; metadata: string } | undefined;
  const after = (db.prepare("SELECT COUNT(*) c FROM episodic_memory_facts WHERE chat_id=?").get(CHAT) as { c: number }).c;
  const inserted = after > before;
  const ev = inserted && row ? (JSON.parse(row.metadata) as { evidence?: unknown }).evidence : null;
  return {
    label: p.label,
    turn: p.turn,
    proposedQuote: p.evidence.source_quote,
    proposedFactText: p.fact.fact_text,
    evidenceType: p.fact.evidence_type,
    parsedValid: parsedSection.valid,
    parsedFactCount: parsedSection.facts?.length ?? null,
    verifierVerdict: verdict,
    inserted: result.inserted,
    storedEvidence: ev,
  };
}
