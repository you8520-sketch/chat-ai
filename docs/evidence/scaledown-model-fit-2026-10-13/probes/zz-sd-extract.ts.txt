const ENTITY_TYPES = {
  promise: "A promise, commitment or agreement to do something in the future, stated by anyone in the text.",
  promise_cancellation: "A statement that cancels, withdraws or revokes an earlier promise or agreement.",
  relationship_proposal: "A proposal, request or confession asking to start or change a romantic relationship.",
  relationship_acceptance: "An explicit statement by the other person that accepts a relationship proposal.",
  past_acquaintance_claim: "A claim that two people knew each other or shared a history before the present scene.",
  emotion: "An emotional state or emotional reaction of a character.",
  npc_action: "An action performed by a third character who is neither the user nor the main character.",
  mentor_or_lover_relationship: "A stated teacher-student or romantic partner relationship between two people.",
};

const INSTRUCTION =
  "This is a roleplay chat log. Extract only spans that literally state the described event. " +
  "Do not infer events that are not written. A claim made by one speaker about the past is not established fact. " +
  "A relationship proposal is not accepted unless the other person explicitly accepts it.";

function sdBody(text: string, withInstruction: boolean) {
  return {
    model: "scaledown/extract",
    messages: [
      ...(withInstruction ? [{ role: "system", content: INSTRUCTION }] : []),
      { role: "user", content: text },
    ],
    provider: { zdr: true, data_collection: "deny", options: { scaledown: { entities: ENTITY_TYPES } } },
  };
}

async function callExtract(text: string, withInstruction: boolean) {
  const started = performance.now();
  const res = await fetch("https://openrouter.ai/api/v1/chat/completions", {
    method: "POST",
    headers: { Authorization: `Bearer ${SD_OR_KEY}`, "Content-Type": "application/json" },
    body: JSON.stringify(sdBody(text, withInstruction)),
  });
  const raw = await res.text();
  const ms = Math.round((performance.now() - started) * 10) / 10;
  let parsed: any = null;
  let content: any = null;
  let parseError: string | null = null;
  try {
    parsed = JSON.parse(raw);
    content = JSON.parse(parsed.choices?.[0]?.message?.content ?? "null");
  } catch (e) {
    parseError = (e as Error).message;
  }
  return { status: res.status, ms, raw, parsed, content, parseError, ...parseCostUsd(raw) };
}

function evalSdEntities(text: string, segments: ReturnType<typeof buildBlocks>["segments"], entities: any[]) {
  return entities.map((e) => {
    const exact = typeof e.start === "number" && typeof e.end === "number" ? text.slice(e.start, e.end) === e.text : false;
    const where =
      typeof e.start === "number" && typeof e.end === "number"
        ? resolveSpan(segments, e.start, e.end)
        : { msgId: null, role: null, crossesBoundary: false, inHeader: false };
    return {
      type: e.type,
      text: e.text,
      confidence: e.confidence,
      start: e.start,
      end: e.end,
      spanMatchesInputSlice: exact,
      msgId: where.msgId,
      role: where.role,
      crossesBoundary: where.crossesBoundary,
      inHeader: where.inHeader,
    };
  });
}

function checkExpect(expect: Expect, evald: ReturnType<typeof evalSdEntities>) {
  const mustHit = expect.mustTypes.map((m) => ({
    ...m,
    hit: evald.some((e) => e.type === m.type && e.role === m.role),
    hitWrongRole: evald.some((e) => e.type === m.type && e.role !== m.role),
  }));
  const mustNotViolations = evald.filter((e) => expect.mustNotTypes.includes(e.type)).map((e) => ({ type: e.type, text: e.text, role: e.role }));
  return { mustHit, mustNotViolations };
}

function judgeLuna(c: Case, run: Awaited<ReturnType<typeof runPostTurnRelationshipOnlyInitial>>) {
  const userBody = c.src.user;
  const asstBody = c.src.assistant;
  const facts = run.parsed?.episodic?.facts ?? [];
  const factJudge = facts.map((f: any) => {
    const q: string = f.evidence?.source_quote ?? "";
    const role: string = f.evidence?.source_role ?? "";
    const inUser = q.length > 0 && userBody.includes(q);
    const inAsst = q.length > 0 && asstBody.includes(q);
    return {
      fact_text: f.fact_text,
      category: f.category,
      evidence_type: f.evidence_type,
      claimedRole: role,
      quote: q,
      quoteExactInClaimedRole: role === "user" ? inUser : role === "assistant" ? inAsst : false,
      quoteFoundInOtherRoleOnly: role === "user" ? !inUser && inAsst : role === "assistant" ? !inAsst && inUser : false,
      quoteNotInAnyMessage: !inUser && !inAsst,
      resolvedMsgId: role === "user" && inUser ? c.src.userMsgId : role === "assistant" && inAsst ? c.src.assistantMsgId : null,
    };
  });
  const rel = run.parsed?.relationship;
  const delta: any = rel && "delta" in rel ? (rel as any).delta : null;
  const removeExact = (delta?.promisesRemove ?? []).map((t: string) => ({ text: t, exactlyMatchesActive: c.activePromises.includes(t) }));
  return {
    transportOk: run.transportOk,
    errorCode: run.errorCode,
    relationshipValid: (rel as any)?.valid ?? null,
    episodicValid: (run.parsed?.episodic as any)?.valid ?? null,
    promisesAdd: delta?.promisesAdd ?? [],
    promisesRemove: removeExact,
    itemsDelta: { items: delta?.items ?? [], itemsRemove: delta?.itemsRemove ?? [] },
    facts: factJudge,
  };
}

async function main() {
  if (!SD_OR_KEY) throw new Error("OPENROUTER_JEV_BENCHMARK_API_KEY required");
  setTapFile(`${OUT}/raw/extract.jsonl`);
  const out: Record<string, unknown> = {
    caseDefs: CASES.map((c) => ({ id: c.id, label: c.label, note: c.note, fixture: c.src.fixture, turn: c.src.turn, activePromises: c.activePromises, expect: c.expect })),
    entityTypes: ENTITY_TYPES,
    instruction: INSTRUCTION,
    lunaModel: BACKGROUND_OPENROUTER_MODEL,
  };

  const results: any[] = [];
  for (const c of CASES) {
    const { text, segments } = buildBlocks([c.src]);
    if (RUN_SD) {
      for (const withInstruction of [false, true]) {
        for (let rep = 1; rep <= REPS; rep++) {
          setTag(`B_sd_${withInstruction ? "x2" : "x1"}_${c.id}_rep${rep}`);
          const r = await callExtract(text, withInstruction);
          const entities = r.content?.entities ?? [];
          const evald = evalSdEntities(text, segments, entities);
          results.push({
            arm: withInstruction ? "scaledown_extract_entities+instruction" : "scaledown_extract_entities_only",
            caseId: c.id,
            rep,
            input: text,
            status: r.status,
            ms: r.ms,
            promptTokens: r.promptTokens,
            completionTokens: r.completionTokens,
            costUsd: r.costUsd,
            parseError: r.parseError,
            rawContent: r.parsed?.choices?.[0]?.message?.content ?? null,
            entities: evald,
            check: checkExpect(c.expect, evald),
          });
        }
      }
    }
    if (RUN_LUNA) {
      for (let rep = 1; rep <= REPS; rep++) {
        setTag(`B_luna_${c.id}_rep${rep}`);
        const before = tapRecords().length;
        const started = performance.now();
        const run = await runPostTurnRelationshipOnlyInitial({
          charName: CHAR,
          personaName: USER_NAME,
          userPersona: PERSONA,
          userMessage: c.src.user,
          assistantProse: c.src.assistant,
          primaryModelId: BACKGROUND_OPENROUTER_MODEL,
          previousAssistantMessage: null,
          activePromises: c.activePromises,
          includeSuggestions: false,
          includeRelationship: true,
          includeEpisodic: true,
        });
        const ms = Math.round((performance.now() - started) * 10) / 10;
        const tapped = tapRecords().slice(before);
        const usd = (run.usage as any)?.cheaperInferenceBilledCostUsd ?? (run.usage as any)?.upstreamCostUsd ?? null;
        results.push({
          arm: "luna_production_shared_initial",
          caseId: c.id,
          rep,
          input: { user: c.src.user, assistant: c.src.assistant, activePromises: c.activePromises },
          ms,
          inputTokens: run.usage?.inputTokens ?? null,
          outputTokens: run.usage?.outputTokens ?? null,
          costUsd: usd,
          physicalCalls: tapped.length,
          rawText: run.text,
          judge: judgeLuna(c, run),
        });
      }
    }
  }
  out.results = results;

  // Window-level Extract (message-ID mapping across 10 messages), ScaleDown only.
  if (RUN_SD) {
    const windows: { id: string; turns: SourceTurn[] }[] = [
      { id: "G8_W_T6_T10", turns: [6, 7, 8, 9, 10].map(g8Turn) },
      { id: "G8_W_T11_T15", turns: [11, 12, 13, 14, 15].map(g8Turn) },
      { id: "G8_W_T21_T25", turns: [21, 22, 23, 24, 25].map(g8Turn) },
      { id: "K1591_W_T16_T20", turns: [16, 17, 18, 19, 20].map(k1591Turn) },
      { id: "K1591_W_T21_T25", turns: [21, 22, 23, 24, 25].map(k1591Turn) },
    ];
    const wres: any[] = [];
    for (const w of windows) {
      const { text, segments } = buildBlocks(w.turns);
      setTag(`B_sd_window_${w.id}`);
      const r = await callExtract(text, false);
      const evald = evalSdEntities(text, segments, r.content?.entities ?? []);
      wres.push({ windowId: w.id, input: text, status: r.status, ms: r.ms, promptTokens: r.promptTokens, costUsd: r.costUsd, rawContent: r.parsed?.choices?.[0]?.message?.content ?? null, entities: evald });
    }
    out.windowResults = wres;
  }

  const usd = tapRecords().filter((r) => r.url.includes("openrouter.ai")).reduce((s, r) => s + (parseCostUsd(r.responseText).costUsd ?? 0), 0);
  const lunaUsd = (results as any[]).filter((r) => r.arm.startsWith("luna")).reduce((s, r) => s + (r.costUsd ?? 0), 0);
  out.tapSummary = { aiCalls: tapRecords().length, openrouterReportedCostUsd: usd, lunaBilledCostUsd: lunaUsd, totalKrw: (usd + lunaUsd) * FX };
  writeJson(`extract-results${process.env.SD_B_SUFFIX ?? ""}.json`, out);
  console.log("B_DONE", JSON.stringify(out.tapSummary));
}

main().catch((e) => {
  console.error("B_FAIL", e);
  process.exit(1);
});
