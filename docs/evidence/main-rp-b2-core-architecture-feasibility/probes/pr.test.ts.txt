function insertTurn(chatId: number, turn: number): Seeded {
  const db = getDb();
  const userId = Number(
    db
      .prepare(`INSERT INTO messages (chat_id, role, content, model) VALUES (?,?,?,?)`)
      .run(chatId, "user", userText(turn), "user").lastInsertRowid
  );
  const assistantId = Number(
    db
      .prepare(
        `INSERT INTO messages (chat_id, role, content, model, user_message_id, generation_status) VALUES (?,?,?,?,?,'completed')`
      )
      .run(chatId, "assistant", assistantText(turn), "test", userId).lastInsertRowid
  );
  return { userId, assistantId };
}

function reconcileTurn(
  chatId: number,
  userId: number,
  charId: number,
  seeded: Seeded,
  turn: number,
  facts: Fact[],
  isRegeneration = false
) {
  return reconcileSharedEpisodicFactsForTurn(getDb(), {
    chatId,
    userId,
    characterId: charId,
    assistantMessageId: seeded.assistantId,
    sourceUserMessageId: seeded.userId,
    sourceUserText: userText(turn),
    userPersona: PERSONA,
    episodic: parseSharedEpisodicSection({ extracted_facts: facts }),
    isRegeneration,
  } as never);
}

function extractiveFake(userContent: string): string {
  const lines = [...userContent.matchAll(/^유저:\s*(.+)$/gm)].map((m) => m[1]!.trim());
  let text = lines.join(" ");
  if (text.length < 80) text = `${text} ${text}`.trim();
  return text;
}

function recentRaw(chatId: number): string {
  const rows = getDb()
    .prepare(`SELECT content FROM messages WHERE chat_id=? ORDER BY id DESC LIMIT 8`)
    .all(chatId) as { content: string }[];
  return rows
    .map((r) => r.content)
    .reverse()
    .join("\n");
}

type QueryResult = { block: string; chars: number; tokens: number };

function runEpisodic(
  chatId: number,
  userId: number,
  charId: number,
  turn: number,
  query: string,
  longTermMemoryText: string
): QueryResult {
  const r = getEpisodicMemoryForPrompt(
    getDb(),
    {
      chatId,
      characterId: charId,
      userId,
      currentTurn: turn + 1,
      currentUserMessage: query,
      recentChatText: recentRaw(chatId),
      longTermMemoryText,
      dynamicMemoryTotalMaxChars: 10_000,
    },
    ENV
  );
  return { block: r.promptBlock, chars: r.promptBlock.length, tokens: estimateTokens(r.promptBlock) };
}

it("compaction: real Luna compactCurrentMemory over G2 summaries + 27 windows", async () => {
  const dump = JSON.parse(readFileSync(`${IN5}/dump-cur-real-luna.json`, "utf8"));
  const seal = JSON.parse(readFileSync("/opt/cursor/artifacts/gate4/seal-A.json", "utf8")) as Record<string, any>;
  const lines: string[] = [];
  for (const r of dump.records as { turnStart: number; turnEnd: number; summary: string }[]) {
    lines.push(`[${r.turnStart}~${r.turnEnd}턴] ${r.summary}`);
  }
  const g2Chars = lines.join("\n").length;
  let t = 81;
  for (const [k, v] of Object.entries(seal)) {
    if (!v.persisted) continue;
    lines.push(`[${t}~${t + 4}턴] ${v.persisted}`);
    t += 5;
  }
  const combined = lines.join("\n");
  const budget = 6000;
  const trace = { requestKind: "compaction-probe" } as never;
  const first = await compactCurrentMemory(combined, budget, trace);
  const KEYS: Record<string, RegExp> = {
    amateur: /아마추어/,
    lipBite: /입술/,
    gazeAvoid: /시선/,
    couple: /연인/,
    breakup: /헤어|이별/,
    sister: /누나/,
    promise: /약속/,
  };
  const survive = (s: string) => Object.fromEntries(Object.entries(KEYS).map(([k, re]) => [k, re.test(s)]));
  out.compaction = {
    g2SummaryChars: g2Chars,
    inputChars: combined.length,
    firstOutputChars: first.length,
    inputSurvive: survive(combined),
    firstSurvive: survive(first),
    firstHasTurnMarkers: /\[\d+~\d+턴\]/.test(first),
    first,
  };
  const second = await compactCurrentMemory(first, 4000, trace);
  (out.compaction as Record<string, unknown>).second = {
    inputChars: first.length,
    outputChars: second.length,
    secondSurvive: survive(second),
    second,
  };
});
