import Module from "module";
const originalLoad = (Module as unknown as { _load: typeof Module._load })._load;
(Module as unknown as { _load: typeof Module._load })._load = function (request: string, parent: NodeModule, isMain: boolean) {
  if (request === "server-only") return {};
  return originalLoad(request, parent, isMain);
} as typeof Module._load;

import { after, before, it } from "node:test";
import { existsSync, readFileSync, writeFileSync } from "node:fs";
import { getDb } from "@/lib/db";
import { callBackgroundMemory } from "@/lib/ai";
import { installIsolatedTestDatabase, uninstallIsolatedTestDatabase } from "@/lib/test/isolatedTestDatabase";
import { getOrCreateChatMemory } from "./memory-db";
import { buildSharedInitialEpisodicSectionInstructions } from "./memory-episodic-prompt";
import { parseSharedEpisodicSection, reconcileSharedEpisodicFactsForTurn } from "./memory-episodic-shared";

const SRC = process.env.SRC ?? "/opt/cursor/artifacts/gate4/sources.json";
const OUTF = process.env.OUTF ?? "/opt/cursor/artifacts/gate5/epi-base.json";
const VARIANT = process.env.VARIANT ?? "base";
const ONLY = (process.env.ONLY ?? "").split(",").filter(Boolean);
const CAP_USD = Number(process.env.CAP_USD ?? 0.06);
const PERSONA = "이름/호칭: 렌\n성격: 조용함";
const CHAR = "라이크";
const sources = JSON.parse(readFileSync(SRC, "utf8")) as Record<string, { kind: string; turns: { user: string; assistant: string }[] }>;
const out: Record<string, any> = existsSync(OUTF) ? JSON.parse(readFileSync(OUTF, "utf8")) : {};
let spent = 0;

const WIDEN = `\nADDITIONAL RULE (continuity): When a character's visibly displayed reaction (an observable action, silence, gaze, posture, tone) to a consequential user line or scene event matters for future relationship continuity, record it as an observable event: explicit_scene_event, subject=the character, source_role=assistant, source_quote=the verbatim reaction clause. Do not write the inferred emotion into any field other than fact_text.`;

function system(): string {
  return [
    "Required sections are enforced by the provider JSON schema on the wire.",
    "Do not include markdown fences or prose outside JSON.",
    'Output exactly one JSON object: {"episodic":{"extracted_facts":[...]}}',
    buildSharedInitialEpisodicSectionInstructions(false) + (VARIANT === "widen" ? WIDEN : ""),
    "Episodic current-turn evidence: current USER message + current canonical ASSISTANT prose only.",
    "Character identity / critical context = canon reference, NOT new episodic evidence.",
  ].join("\n");
}

before(() => {
  installIsolatedTestDatabase();
  process.env.MEMORY_FEATURE_ENABLED = "1";
});
after(() => {
  writeFileSync(OUTF, JSON.stringify({ variant: VARIANT, spent, windows: out }, null, 2));
  uninstallIsolatedTestDatabase();
});

let n = 0;
for (const [id, src] of Object.entries(sources)) {
  if (ONLY.length && !ONLY.includes(id)) continue;
  if (out[id]) continue;
  const idx = ++n;
  it(`${VARIANT} ${id}`, async () => {
    if (spent > CAP_USD) { out[id] = { skipped: "CAP" }; return; }
    const db = getDb();
    const userId = 960000 + idx * 10, chatId = userId + 1, charId = userId + 2;
    db.prepare(`INSERT INTO users (id, email, nickname, pw_hash) VALUES (?,?,?,?)`).run(userId, `e-${userId}@t.l`, "렌", "x");
    db.prepare(`INSERT INTO characters (id, name) VALUES (?,?)`).run(charId, CHAR);
    db.prepare(`INSERT INTO chats (id, user_id, character_id, mode) VALUES (?,?,?,'safe')`).run(chatId, userId, charId);
    getOrCreateChatMemory(chatId, userId, charId, "pro");
    db.prepare(`INSERT INTO messages (chat_id, role, content, model) VALUES (?,?,?,?)`).run(chatId, "assistant", "인사.", "greeting");
    const perTurn: any[] = [];
    for (let t = 0; t < src.turns.length; t++) {
      const turn = src.turns[t]!;
      const uid = Number(db.prepare(`INSERT INTO messages (chat_id, role, content, model) VALUES (?,?,?,?)`).run(chatId, "user", turn.user, "user").lastInsertRowid);
      const aid = Number(db.prepare(`INSERT INTO messages (chat_id, role, content, model, user_message_id, generation_status) VALUES (?,?,?,?,?,'completed')`).run(chatId, "assistant", turn.assistant, "test", uid).lastInsertRowid);
      const r = await callBackgroundMemory(system(), [{ role: "user", content: `[THIS TURN — USER]\n${turn.user}\n\n[THIS TURN — ASSISTANT]\n${turn.assistant}` }], undefined, "background-memory-extract", { maxTokens: 1500 });
      const usd = (r.usage as unknown as { cheaperInferenceBilledCostUsd?: number }).cheaperInferenceBilledCostUsd ?? 0;
      spent += usd;
      let parsed: any = null;
      try { parsed = JSON.parse(r.text.replace(/^```(?:json)?\s*|\s*```$/g, "").trim()); } catch { parsed = null; }
      const section = parseSharedEpisodicSection(parsed?.episodic);
      let result: unknown = { skipped: true, reason: "section_invalid" };
      if (section.valid) {
        result = reconcileSharedEpisodicFactsForTurn(db, {
          chatId, userId, characterId: charId, assistantMessageId: aid, sourceUserMessageId: uid,
          sourceUserText: turn.user, userPersona: PERSONA, episodic: section, isRegeneration: false,
        } as never);
      }
      perTurn.push({ turn: t + 1, usd, rawFacts: parsed?.episodic?.extracted_facts ?? null, sectionValid: section.valid, sanitizedCount: section.facts.length, result });
    }
    const rows = db.prepare(`SELECT source_turn, category, subject, attribute, fact_text, metadata FROM episodic_memory_facts WHERE chat_id=? ORDER BY source_turn, id`).all(chatId) as any[];
    out[id] = { kind: src.kind, perTurn, rows: rows.map((r) => ({ ...r, metadata: JSON.parse(r.metadata ?? "{}") })) };
    writeFileSync(OUTF, JSON.stringify({ variant: VARIANT, spent, windows: out }, null, 2));
  });
}
