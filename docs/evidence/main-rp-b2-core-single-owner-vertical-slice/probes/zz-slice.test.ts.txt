  request: string,
  parent: NodeModule,
  isMain: boolean

  const db = getDb();
  const userId = ++SEQ;
  const charId = ++SEQ;
  const chatId = ++SEQ;
  db.prepare(`INSERT INTO users (id, email, nickname, pw_hash) VALUES (?,?,?,?)`).run(
    userId,
    `s-${userId}@test.local`,
    "렌",
    "x"
  );
  db.prepare(`INSERT INTO characters (id, name) VALUES (?,?)`).run(charId, "라이크");
  db.prepare(`INSERT INTO chats (id, user_id, character_id, mode) VALUES (?,?,?,'safe')`).run(
    chatId,
    userId,
    charId
  );
  getOrCreateChatMemory(chatId, userId, charId, "pro");
  db.prepare(`INSERT INTO messages (chat_id, role, content, model) VALUES (?,?,?,?)`).run(
    chatId,
    "assistant",
    "인사.",
    "greeting"
  );
  return { chatId, userId, charId };
}

function insertTurn(chatId: number, user: string, assistant: string) {
  const db = getDb();
  const userMid = Number(
    db
      .prepare(`INSERT INTO messages (chat_id, role, content, model) VALUES (?,?,?,?)`)
      .run(chatId, "user", user, "user").lastInsertRowid
  );
  const assistantMid = Number(
    db
      .prepare(
        `INSERT INTO messages (chat_id, role, content, model, user_message_id, generation_status) VALUES (?,?,?,?,?,'completed')`
      )
      .run(chatId, "assistant", assistant, "test", userMid).lastInsertRowid
  );
  return { userMid, assistantMid };
}

type Label = EpisodicExtractedFact["evidence_type"];
function mk(
  p: {
    category: EpisodicExtractedFact["category"];
    subject: string;
    gloss: string;
    type: Label;
    role: "user" | "assistant";
    quote: string;
  }
): EpisodicExtractedFact {
  return {
    category: p.category,
    subject: p.subject,
    attribute: `attr_${p.quote.length}_${p.gloss.length}`,
    value: `val_${p.gloss.length}_${p.quote.length}`,
    importance: "important",
    fact_text: p.gloss,
    evidence_type: p.type,
    evidence: { source_role: p.role, source_quote: p.quote },
  } as EpisodicExtractedFact;
}

it("V: evidence verifier matrix on the live gate (honest / plausibly-wrong / evasive labels)", () => {
  const { chatId } = seedChat();
  const rows: unknown[] = [];
  for (const c of CASES) {
    const t = insertTurn(chatId, c.user, c.assistant);
    const sources = loadEpisodicEvidenceSources(getDb(), chatId, {
      userMessageId: t.userMid,
      assistantMessageId: t.assistantMid,
    });
    for (const [variant, fact] of Object.entries(c.variants)) {
      const d = evaluateEpisodicFactEvidence(fact, sources, { userPersona: PERSONA });
      rows.push({
        case: c.id,
        title: c.title,
        variant,
        labels: { type: fact.evidence_type, category: fact.category, subject: fact.subject },
        gloss: fact.fact_text,
        accepted: d.accepted,
        reason: d.accepted ? null : d.reason,
        authority: d.accepted ? d.evidence.authority : null,
        wireLine: d.accepted ? renderVerifiedEvidenceText(d.evidence) : null,
        falseContent: c.falseContent,
        glossOnWire: false,
      });
    }
  }
  out.verifierMatrix = rows;
});

const P_TEXT = "내일까지 검을 갈아다 준다";

async function applyDelta(
  chatId: number,
  turn: { userMid: number; assistantMid: number },
  userMessage: string,
  assistantMessage: string,
  payload: Record<string, unknown>
) {
  const delta = normalizeRelationshipMetaDeltaFromJson(payload, userMessage, NAMES);
  await mergeRelationshipMetaFromTurn({
    chatId,
    names: NAMES,
    userMessage,
    assistantMessage,
    route: "safe",
    sharedInitialAttempted: true,
    sharedInitialDelta: delta,
    sharedInitialParsed: true,
    sourceUserMessageId: turn.userMid,
    assistantMessageId: turn.assistantMid,
  } as never);
}

function promises(chatId: number): string[] {
  return loadChatRelationshipMeta(chatId, NAMES).promises.map((p) => p.text);
}

// LP. Prototype of the ledger leg (probe-only, NOT product code): authority is derived from the
// role of the canonical message that contains the quote; the LLM supplies a quote only, never a label,
// and its gloss never reaches the wire. Legacy entries (no evidence) are never promoted.
// ---------------------------------------------------------------------------
type ProtoEv = { role: "user" | "assistant"; mid: number; quote: string; label: string };
function protoDeriveEvidence(chatId: number, binding: { userMid: number; assistantMid: number }, quote: string): ProtoEv | null {
  const sources = loadEpisodicEvidenceSources(getDb(), chatId, {
    userMessageId: binding.userMid,
    assistantMessageId: binding.assistantMid,
  });
  const q = norm(quote);
  if ([...q.replace(/\s+/gu, "")].length < 6 || [...q].length > 300) return null;
  const inUser = sources.user?.text.includes(q) ?? false;
  const inAssistant = sources.assistant?.text.includes(q) ?? false;
  if (!inUser && !inAssistant) return null;
  if (inUser && !inAssistant) return { role: "user", mid: sources.user!.id, quote: q, label: "유저가 직접 쓴 원문" };
  return { role: "assistant", mid: sources.assistant!.id, quote: q, label: "AI가 쓴 원문(사실 미확정)" };
}
function protoLine(turn: number, ev: ProtoEv): string {
  return `- [T${turn}] [${ev.label}] “${ev.quote}”`;
}
