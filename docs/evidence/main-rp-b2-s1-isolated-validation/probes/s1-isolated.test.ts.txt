function dialogueOf(window: RealLunaWindow): string {
  return window.turns
    .map((t, i) => `[${window.turnStart + i}턴]\n유저: ${t.user}\n${CHAR_NAME}: ${t.assistant}`)
    .join("\n\n");
}

function cleanup() {
  const db = getDb();
  for (const chatId of [CHAT, CHAT_FORK]) {
    db.prepare("DELETE FROM episodic_memory_facts WHERE chat_id=?").run(chatId);
    db.prepare("DELETE FROM chat_turn_summaries WHERE chat_id=?").run(chatId);
    db.prepare("DELETE FROM chat_memories WHERE chat_id=?").run(chatId);
    db.prepare("DELETE FROM messages WHERE chat_id=?").run(chatId);
    db.prepare("DELETE FROM chats WHERE id=?").run(chatId);
  }
  db.prepare("DELETE FROM users WHERE id=?").run(USER);
  db.prepare("DELETE FROM characters WHERE id=?").run(CHAR);
}

type Stub = (windowStart: number, attempt: number) => string | Error;

function installStub(stub: Stub) {
  const callsPerStart = new Map<number, number>();
  let totalCalls = 0;
  rolling.__setSummarizeTurnBatchCallerForTests(async (_system, history) => {
    const prompt = history.map((m) => m.content).join("\n");
    const m = prompt.match(/\[(\d+)~(\d+)턴 원본 대화\]/);
    const start = Number(m![1]);
    const attempt = callsPerStart.get(start) ?? 0;
    callsPerStart.set(start, attempt + 1);
    totalCalls += 1;
    const result = stub(start, attempt);
    if (result instanceof Error) throw result;
    return { text: result };
  });
  return { callsPerStart, calls: () => totalCalls };
}

async function sealAll(opts: {
  persona?: string;
  rounds?: number;
  allowUnverified: boolean;
}): Promise<number> {
  let sealedRounds = 0;
  for (let round = 0; round < (opts.rounds ?? TOTAL + 4); round++) {
    const sealed = await rolling.processRollingSummaryBatch({
      chatId: CHAT,
      userId: USER,
      characterId: CHAR,
      charName: CHAR_NAME,
      userPersona: opts.persona,
      tier: "pro",
      memoryCapacity: MEMORY_CAPACITY_FIXED,
      allowUnverifiedSeal: opts.allowUnverified,
    } as any);
    if (!sealed) break;
    sealedRounds += 1;
  }
  return sealedRounds;
}

function recordsOf() {
  return turnSummary.listMemoryRecordsForChat(CHAT).filter((r) => !r.inactive);
}
function labelOf(r: { sealAuthority?: string | null }): string {
  return (r as any).sealAuthority ?? "normal";
}
function frontierOf(turns: number): number {
  return integrity.highestContiguousCompletedTurn(recordsOf(), turns);
}

async function buildWire(windowCount: number, persona: string | undefined) {
  const turns = windowCount * 5;
  const records = recordsOf();
  const summarized = frontierOf(turns);
  const rows = getDb()
    .prepare(`SELECT role, content, model FROM messages WHERE chat_id=? ORDER BY id`)
    .all(CHAT) as { role: "user" | "assistant"; content: string; model?: string }[];
  const allTurns = messagesToTurns(rows);
  const rawPool = resolveProviderRawPoolExchangeCount({
    memoryFeatureEnabled: true,
    completedTurns: turns,
    summarizedTurnCount: summarized,
  });
  const rawHistory = rawRecentTurnsToHistory(allTurns, rawPool, {
    summarizedTurnCount: summarized,
    memoryFeatureEnabled: true,
  });
  const excludeTurnStartGte = resolveLorebookExcludeFromTrimmedHistory(allTurns, rawHistory);
  const injection = await buildMemoryContextForChat({
    chatId: CHAT,
    userId: USER,
    characterId: CHAR,
    tier: "pro",
    memoryCapacity: MEMORY_CAPACITY_FIXED,
    userMessage: "다음 장면으로 넘어간다.",
    excludeSummaryTurnStartGte: excludeTurnStartGte,
    modelId: "deepseek-v4.1-flash",
    provider: "cheaperinference",
  });
  const unverified = buildUnverifiedSealBlock
    ? buildUnverifiedSealBlock({ chatId: CHAT, excludeTurnStartGte })
    : { text: "", chars: 0, blockCount: 0 };
  const built = buildContext({
    charName: CHAR_NAME,
    chunks: [],
    userNickname: USER_NAME,
    longTermMemory: injection.text,
    mediumTermMemoryBlock: injection.mediumTermText,
    unverifiedSummaryBlock: unverified.text,
    archiveMemory: injection.archiveText,
    shortTermHistory: rawHistory,
    currentUserMessage: "다음 장면으로 넘어간다.",
    nsfw: false,
    provider: "cheaperinference",
    modelId: "deepseek-v4.1-flash",
    completedTurns: turns,
    summarizedTurnCount: summarized,
    chatId: CHAT,
    userId: USER,
  } as any);
  const rawHistoryChars = rawHistory.reduce((n, m) => n + String(m.content).length, 0);
  const sections = built.meta.trackedSections ?? [];
  const sec = (id: string) => sections.find((s: any) => s.id === id)?.text ?? "";
  const systemPrompt = built.systemPrompt as string;
  const ltmRegion = systemPrompt.match(/<LONG_TERM_MEMORY>[\s\S]*?<\/LONG_TERM_MEMORY>/)?.[0] ?? "";
  return {
    records,
    summarized,
    excludeTurnStartGte,
    current: sec("current-memory"),
    medium: sec("medium-term-memory"),
    unverifiedSection: sec("unverified-summary"),
    ltmRegion,
    systemPrompt,
    estimatedSystemTokens: (built.meta as any).estimatedSystemTokens as number,
    wrapperPresent: ltmRegion.length > 0,
    rawHistoryMessages: rawHistory.length,
    rawHistoryChars,
    rawPool,
    injection,
    unverified,
  };
}

function locate(wire: Awaited<ReturnType<typeof buildWire>>, windowCount: number) {
  const out: Record<string, string> = {};
  for (const e of EVENTS) {
    if (e.start + 4 > windowCount * 5) continue;
    const inLtm = wire.ltmRegion.includes(e.needle);
    const inUnverified = wire.unverifiedSection.includes(e.needle);
    const inCurrent = wire.current.includes(e.needle) || wire.medium.includes(e.needle);
    const windowSealed = wire.records.some((r) => r.turnStart === e.start);
    const rawOwned = e.start >= wire.excludeTurnStartGte;
    out[e.key] = inLtm
      ? "established_past(LONG_TERM_MEMORY)"
      : inUnverified
        ? "unverified_block(outside wrapper)"
        : inCurrent
          ? "memory_section_no_wrapper"
          : rawOwned
            ? "raw_history_window"
            : windowSealed
              ? "sealed_but_absent_from_wire"
              : "not_sealed(frontier_stalled)";
  }
  return out;
}
