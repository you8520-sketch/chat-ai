import { readFileSync, writeFileSync } from "node:fs";

const fixture = JSON.parse(readFileSync("/tmp/real-luna.json", "utf8"));
const turns = [];
for (const w of fixture.windows) {
  w.turns.forEach((t, i) => {
    turns.push({
      n: w.turnStart + i,
      windowStart: w.turnStart,
      user: t.user,
      assistant: t.assistant,
    });
  });
}
const byTurn = new Map(turns.map((t) => [t.n, t]));

function roleText(turn, role) {
  return role === "user" ? turn.user : turn.assistant;
}

let seq = 1;
function acceptOps(ops, windowStart, windowEnd, prior) {
  const accepted = [];
  const rejected = [];
  for (const op of ops ?? []) {
    const quote = String(op.quote ?? "");
    const role = op.role === "assistant" ? "assistant" : op.role === "user" ? "user" : "";
    const turnN = Number(op.turn);
    const turn = byTurn.get(turnN);
    const inWindow = turn && turnN >= windowStart && turnN <= windowEnd;
    const quoteOk = Boolean(role && turn && quote && roleText(turn, role).includes(quote));
    if (!inWindow || !quoteOk) {
      rejected.push({ op, reason: !inWindow ? "turn_outside_window" : "quote_role_mismatch" });
      continue;
    }
    const kind = String(op.kind ?? "utterance");
    if (kind === "relationship_established" || kind === "lovers" || op.op === "establish") {
      rejected.push({ op, reason: "ambiguous_acceptance_not_promoted" });
      continue;
    }
    if (op.op === "withdraw") {
      const target = prior.find((e) => e.id === op.targetId);
      if (!target || role !== "user") {
        rejected.push({ op, reason: "withdraw_target_or_role" });
        continue;
      }
      accepted.push({
        id: `w${seq++}`,
        op: "withdraw_utterance",
        kind: "user_utterance",
        turn: turnN,
        role,
        quote,
        targetId: target.id,
        status: "not_auto_closed",
      });
      continue;
    }
    accepted.push({
      id: `e${seq++}`,
      op: "add",
      kind: kind === "character_claim" ? "character_claim" : role === "assistant" ? "assistant_scene" : "user_utterance",
      turn: turnN,
      role,
      quote,
      status: "open",
    });
  }
  return { accepted, rejected };
}

function assert(cond, msg) {
  if (!cond) throw new Error(msg);
}

// Deterministic gate, no network.
{
  const prior = [{ id: "e-promise", turn: 2, role: "user", quote: "내일까지 검을 갈아다 줄게. 약속이야.", kind: "user_utterance", status: "open" }];
  const bad = acceptOps(
    [
      { op: "add", kind: "relationship_established", turn: 10, role: "assistant", quote: "손을 맞잡았다" },
      { op: "add", kind: "user_utterance", turn: 10, role: "user", quote: "렌과 라이크는 장작을 쌓았다" },
      { op: "add", kind: "character_claim", turn: 24, role: "assistant", quote: "누나가 시장에서 널 알아봤대." },
      { op: "withdraw", targetId: "e-promise", turn: 15, role: "user", quote: "그 약속은 취소할게." },
      { op: "add", kind: "user_utterance", turn: 2, role: "user", quote: "내일까지 검을 갈아다 줄게. 약속이야." },
    ],
    6,
    40,
    prior
  );
  assert(bad.rejected.some((r) => r.reason === "ambiguous_acceptance_not_promoted"), "handhold");
  assert(bad.rejected.some((r) => r.reason === "quote_role_mismatch"), "joint wood");
  assert(bad.accepted.some((a) => a.kind === "character_claim" && a.quote.includes("누나")), "sister claim");
  assert(bad.accepted.some((a) => a.op === "withdraw_utterance" && a.status === "not_auto_closed"), "cancel not closed");
  assert(bad.rejected.some((r) => r.reason === "turn_outside_window"), "promise turn outside later window");
  const promiseStillOpen = prior[0].status === "open";
  assert(promiseStillOpen, "prior promise stays open");
}

const key = process.env.CHEAPER_INFERENCE_BENCHMARK_API_KEY?.trim();
if (!key) {
  console.error("NO_BENCHMARK_KEY");
  process.exit(2);
}

const targets = [1, 6, 11, 16, 21, 36];
const state = [];
const rows = [];
let billed = 0;
const KRW_PER_USD = 1367.698;
const CAP_USD = 1000 / KRW_PER_USD;

async function luna(system, user) {
  const res = await fetch("https://api.cheaperinference.com/v1/chat/completions", {
    method: "POST",
    headers: {
      "Content-Type": "application/json",
      Authorization: `Bearer ${key}`,
    },
    body: JSON.stringify({
      model: "gpt-6-luna",
      temperature: 0,
      max_tokens: 900,
      messages: [
        { role: "system", content: system },
        { role: "user", content: user },
      ],
    }),
  });
  const json = await res.json();
  if (!res.ok) {
    const err = new Error(`HTTP ${res.status}`);
    err.body = json;
    throw err;
  }
  const text = json.choices?.[0]?.message?.content ?? "";
  const usage = json.usage ?? {};
  const cost = Number(usage.cost ?? usage.total_cost ?? json.cost ?? 0) || 0;
  return { text, usage, cost, id: json.id ?? null };
}

function parseJson(text) {
  const fenced = text.match(/```(?:json)?\s*([\s\S]*?)```/);
  const raw = (fenced ? fenced[1] : text).trim();
  return JSON.parse(raw);
}

const system = `당신은 하브의 기존 5턴 기억 작성자다. 이전 기억 상태와 이번 5턴 원문을 함께 보고 JSON만 출력한다.
narrative는 이번 구간의 감정, NPC, 장면 진행을 3인칭으로 적는다. 원문에 없는 관계는 쓰지 않는다.
ops는 이번 구간 원문에 글자 그대로 있는 quote만 담는다.
add는 새 발화나 장면이다. withdraw는 유저가 이전 상태의 약속을 거두어들이는 말이 있을 때만 쓰고 targetId를 이전 id로 둔다.
kind는 user_utterance, assistant_scene, character_claim 중 하나다. 관계 성립을 kind로 쓰지 않는다.
{"narrative":"...","ops":[{"op":"add"|"withdraw","kind":"...","turn":1,"role":"user"|"assistant","quote":"...","targetId":"..."}]}`;

for (const start of targets) {
  if (billed > CAP_USD) throw new Error("cost cap");
  const windowTurns = turns.filter((t) => t.windowStart === start);
  const end = windowTurns.at(-1).n;
  const dialogue = windowTurns
    .map((t) => `[${t.n}턴]\n유저: ${t.user}\n라이크: ${t.assistant}`)
    .join("\n\n");
  const prior = state
    .filter((e) => e.op === "add")
    .map((e) => ({ id: e.id, turn: e.turn, role: e.role, kind: e.kind, quote: e.quote, status: e.status }));
  const user = `[이전 기억 상태]\n${JSON.stringify(prior)}\n\n[이번 ${start}~${end}턴 원문]\n${dialogue}`;
  const call = await luna(system, user);
  billed += call.cost;
  let parsed = null;
  let parseError = null;
  try {
    parsed = parseJson(call.text);
  } catch (e) {
    parseError = String(e.message ?? e);
  }
  const gated = parsed ? acceptOps(parsed.ops, start, end, state) : { accepted: [], rejected: [] };
  state.push(...gated.accepted);
  rows.push({
    window: `${start}-${end}`,
    usage: call.usage,
    cost: call.cost,
    providerId: call.id,
    parseError,
    narrative: parsed?.narrative ?? null,
    raw: call.text,
    accepted: gated.accepted,
    rejected: gated.rejected,
  });
  console.log(start, "accepted", gated.accepted.length, "rejected", gated.rejected.length, "cost", call.cost, "billed", billed);
}

const wire = state
  .map((e) => `[T${e.turn}] ${e.role} ${e.kind} ${e.status}${e.targetId ? ` -> ${e.targetId}` : ""}: ${e.quote}`)
  .join("\n");
const narratives = rows.map((r) => r.narrative).filter(Boolean).join("\n");
const flags = (text) => ({
  promise: text.includes("내일까지 검을 갈아다 줄게. 약속이야."),
  cancel: text.includes("그 약속은 취소할게"),
  proposal: text.includes("좋아, 우리 오늘부터 연인하자."),
  handhold: text.includes("손을 맞잡"),
  falseStart: text.includes("연인 관계를 시작"),
  falseJoint: text.includes("차례로 쌓으며") || text.includes("렌과 라이크는 장작"),
  breakup: text.includes("미안, 우리 그만 헤어지자."),
  seorinBow: text.includes("목례"),
  amateur: text.includes("아마추어"),
  lip: text.includes("입술을 깨물"),
  sister: text.includes("누나가 시장에서"),
  mentor: text.includes("스승"),
  lifelong: text.includes("평생"),
});

const out = {
  model: "gpt-6-luna",
  calls: rows.length,
  billedUsd: billed,
  billedKrw: Math.round(billed * KRW_PER_USD * 10) / 10,
  stateChars: wire.length,
  narrativeChars: narratives.length,
  stateFlags: flags(wire),
  narrativeFlags: flags(narratives),
  promiseStillOpen: !state.some((e) => e.status === "cancelled" || e.status === "closed"),
  withdrawals: state.filter((e) => e.op === "withdraw_utterance"),
  wire,
  narratives,
  rows: rows.map((r) => ({ ...r, raw: r.raw.slice(0, 4000) })),
};
writeFileSync("/tmp/state-proto-results.json", JSON.stringify(out, null, 2));
console.log(JSON.stringify({ billed, krw: out.billedKrw, stateFlags: out.stateFlags, narrativeFlags: out.narrativeFlags, stateChars: wire.length }, null, 2));
