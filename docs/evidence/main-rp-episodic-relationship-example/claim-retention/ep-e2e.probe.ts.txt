let seq = 0;
before(() => {
  installIsolatedTestDatabase();
  const db = getDb();
  db.prepare(`INSERT INTO users (id, email, nickname, pw_hash) VALUES (?,?,?,?)`).run(USER, `e2e-${USER}@test.local`, "e2e", "x");
  db.prepare(`INSERT INTO characters (id, name) VALUES (?,?)`).run(CHAR, "라이크");
});
after(() => uninstallIsolatedTestDatabase());

it("e2e probe", async () => {
  const db = getDb();
  for (const c of CASES.filter((x) => !ONLY || x.id === ONLY)) {
    for (let rep = 0; rep < REPS; rep++) {
      const file = `${OUT}/${ARM}__${c.id}__${rep}.json`;
      if (existsSync(file)) continue;
      const res = await runPostTurnSharedInitial({
        mode: "relationship_only", charName: "라이크", personaName: "렌", userPersona: "이름/호칭: 렌",
        userMessage: c.user, assistantProse: c.assistant, primaryModelId: BACKGROUND_OPENROUTER_MODEL,
        includeSuggestions: false, includeRelationship: true, includeEpisodic: true, relationshipRegenContext: null, activePromises: [],
      } as any);
      seq += 1;
      const chatId = 920200 + seq, um = chatId * 10 + 1, am = chatId * 10 + 2;
      db.prepare(`INSERT INTO chats (id, user_id, character_id, mode) VALUES (?,?,?,'safe')`).run(chatId, USER, CHAR);
      getOrCreateChatMemory(chatId, USER, CHAR, "free");
      db.prepare(`INSERT INTO messages (id, chat_id, role, content) VALUES (?,?,?,?)`).run(um, chatId, "user", c.user);
      db.prepare(`INSERT INTO messages (id, chat_id, role, content, user_message_id, generation_status, model) VALUES (?,?,?,?,?,?,?)`).run(am, chatId, "assistant", c.assistant, um, "completed", "test-model");
      let reconcileErr: string | null = null;
      let inserted = 0;
      try {
        const r = reconcileSharedEpisodicFactsForTurn(db, {
          chatId, userId: USER, characterId: CHAR, assistantMessageId: am, sourceUserMessageId: um, sourceUserText: c.user,
          userPersona: "이름/호칭: 렌", episodic: res.parsed!.episodic, isRegeneration: false,
        } as any);
        inserted = (r as any).inserted ?? 0;
      } catch (e) { reconcileErr = String((e as Error).message); }
      const rows = db.prepare("SELECT category, subject, attribute, value, fact_text, metadata FROM episodic_memory_facts WHERE chat_id=? ORDER BY id").all(chatId) as any[];
      const stored = rows.map((r) => { const ev = (JSON.parse(r.metadata) as any).evidence ?? null; return { category: r.category, subject: r.subject, attribute: r.attribute, value: r.value, fact_text: r.fact_text, evidence: ev }; });
      const wire = getEpisodicMemoryForPrompt(db, { chatId, characterId: CHAR, userId: USER, currentTurn: 80, currentUserMessage: c.query, recentChatText: "평범한 최근 대화" }, { EPISODIC_MEMORY_RECALL_ENABLED: "1" } as NodeJS.ProcessEnv).promptBlock;
      const wireAccept = getEpisodicMemoryForPrompt(db, { chatId, characterId: CHAR, userId: USER, currentTurn: 80, currentUserMessage: "라이크가 연인 제안을 수락했어?", recentChatText: "평범한 최근 대화" }, { EPISODIC_MEMORY_RECALL_ENABLED: "1" } as NodeJS.ProcessEnv).promptBlock;
      const billed = (res.usage as any)?.cheaperInferenceBilledCostUsd ?? null;
      writeFileSync(file, JSON.stringify({ arm: ARM, case: c, rep, model: BACKGROUND_OPENROUTER_MODEL, transportOk: res.transportOk, usage: res.usage, billed, emitted: res.parsed?.episodic?.facts ?? [], episodicValid: res.parsed?.episodic?.valid ?? null, inserted, reconcileErr, stored, wire, wireAccept, rawText: res.text }, null, 1));
    }
  }
});
