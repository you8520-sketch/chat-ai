import Module from "module";
const o = (Module as any)._load;
(Module as any)._load = function (r: string, p: any, m: any) { if (r === "server-only") return {}; return o(r, p, m); };
import { existsSync, mkdirSync, writeFileSync } from "node:fs";
import { it } from "node:test";
import { BACKGROUND_OPENROUTER_MODEL } from "@/lib/ai";
import { runPostTurnSharedInitial } from "@/lib/postTurnSharedInitial/run";
import { buildPostTurnSharedInitialSystem, buildPostTurnSharedInitialUserBlock } from "@/lib/postTurnSharedInitial/prompt";
import { evaluateEpisodicFactEvidence, renderVerifiedEvidenceText } from "./memory-episodic-evidence";

const LABEL = process.env.EP_LABEL ?? "before";
const REPS = Number(process.env.EP_REPS ?? 4);
const OUT = `/tmp/ep/raw`;
mkdirSync(OUT, { recursive: true });
type Case = { id: string; user: string; assistant: string };
const CASES: Case[] = [
  { id: "FX06_proposal_only", user: "좋아, 우리 오늘부터 연인하자.", assistant: "라이크는 놀란 얼굴로 한참 말이 없다가 손을 맞잡았다." },
  { id: "CTL_explicit_accept", user: "좋아, 우리 오늘부터 연인하자.", assistant: "라이크는 한참 말이 없다가 입꼬리를 올렸다. \"그래. 오늘부터 연인이다.\"" },
  { id: "CTL_char_claim", user: "시장에 다녀왔어.", assistant: "라이크는 고개를 끄덕였다. \"누나가 시장에서 널 알아봤대. 예전부터 알던 사이라더라.\"" },
  { id: "CTL_emotion_npc", user: "서린이 건넨 꿀차를 고맙게 받아 마신다.", assistant: "서린은 렌에게만 꿀차를 한 잔 더 따라 주었다. 라이크는 말없이 그 모습을 지켜보다 칼집을 움켜쥐고, 퉁명스럽게 자리를 재촉했다. \"먹었으면 가자.\"" },
  { id: "CTL_user_promise", user: "내일까지 검을 갈아다 줄게. 약속이야.", assistant: "라이크는 고개를 끄덕이며 작게 웃었다." },
];
const MAX_USD = Number(process.env.EP_MAX_USD ?? 0.05);
let usd = 0, calls = 0;

it("episodic prompt live probe", async () => {
  for (const c of CASES.filter((x) => !process.env.EP_ONLY || x.id === process.env.EP_ONLY)) {
    for (let rep = 0; rep < REPS; rep++) {
      const file = `${OUT}/${LABEL}__${c.id}__${rep}.json`;
      if (existsSync(file)) continue;
      if (usd >= MAX_USD) { console.log("CAP"); return; }
      const input = {
        mode: "relationship_only" as const, charName: "라이크", personaName: "렌", userPersona: "이름/호칭: 렌",
        userMessage: c.user, assistantProse: c.assistant, primaryModelId: BACKGROUND_OPENROUTER_MODEL,
        includeSuggestions: false, includeRelationship: true, includeEpisodic: true, relationshipRegenContext: null, activePromises: [] as string[],
      };
      calls++;
      const res = await runPostTurnSharedInitial(input);
      const billed = (res.usage as any)?.cheaperInferenceBilledCostUsd ?? (res.usage as any)?.upstreamCostUsd ?? null;
      if (typeof billed === "number") usd += billed;
      const facts = res.parsed?.episodic?.facts ?? [];
      const sources = { user: { id: 1, text: c.user }, assistant: { id: 2, text: c.assistant } } as any;
      const verdicts = facts.map((f: any) => {
        const d = evaluateEpisodicFactEvidence(f, sources, { userPersona: "이름/호칭: 렌" });
        return { fact: f, accepted: d.accepted, reason: (d as any).reason ?? null, injected: d.accepted ? renderVerifiedEvidenceText(d.evidence) : null };
      });
      writeFileSync(file, JSON.stringify({ label: LABEL, case: c, rep, model: BACKGROUND_OPENROUTER_MODEL, ok: res.transportOk, usage: res.usage, billed, episodicPresent: res.parsed?.episodic?.present ?? null, episodicValid: res.parsed?.episodic?.valid ?? null, verdicts, text: res.text }, null, 1));
    }
  }
  const sys = buildPostTurnSharedInitialSystem({ mode: "relationship_only", charName: "라이크", personaName: "렌", userMessage: "x", assistantProse: "y", primaryModelId: BACKGROUND_OPENROUTER_MODEL, includeSuggestions: false, includeRelationship: true, includeEpisodic: true, relationshipRegenContext: null, activePromises: [] } as any);
  writeFileSync(`${OUT}/${LABEL}__SYSTEM_PROMPT.txt`, sys);
  console.log("DONE", calls, usd);
});
