import json,glob,sys,collections
label=sys.argv[1]
rows=collections.defaultdict(list)
for f in sorted(glob.glob(f'/tmp/ep/raw/{label}__*.json')):
    d=json.load(open(f)); rows[d['case']['id']].append(d)
tot_in=tot_out=0;usd=0
for cid,ds in rows.items():
    print('==',cid,len(ds))
    for d in ds:
        u=d['usage'] or {}
        tot_in+=u.get('inputTokens',0) or 0; tot_out+=u.get('outputTokens',0) or 0; usd+=d['billed'] or 0
        if not d['verdicts']: print('  rep',d['rep'],'no facts')
        for v in d['verdicts']:
            f=v['fact']
            print('  rep',d['rep'],'ACC' if v['accepted'] else 'REJ:'+str(v['reason']),'|',f['category'],f['subject'],f['attribute'],f['value'],'|',f['fact_text'],'|',f['evidence_type'],f['evidence']['source_role'],'"'+f['evidence']['source_quote'][:30]+'"','=>',v['injected'])
print('tokens in',tot_in,'out',tot_out,'usd',round(usd,6))
