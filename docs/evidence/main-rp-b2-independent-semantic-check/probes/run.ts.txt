function dialogue(start: number): string {
  const w = fx.windows.find((x: any) => x.turnStart === start);
  return w.turns
    .map((t: any, i: number) => `[${start + i}턴]\n유저: ${t.user}\n라이크: ${t.assistant}`)
    .join("\n\n");
}
function luna(start: number): string {
  return fx.windows.find((x: any) => x.turnStart === start).lunaOutput;
}

function parseJson(text: string): any {
  const fenced = text.match(/```(?:json)?\s*([\s\S]*?)```/);
  const raw = (fenced?.[1] ?? text).trim();
  const start = raw.indexOf("{");
  const end = raw.lastIndexOf("}");
  if (start < 0 || end < start) throw new Error("no json");
  return JSON.parse(raw.slice(start, end + 1));
}

async function one(c: (typeof cases)[number]) {
  const user = `[구간 원본]\n${dialogue(c.window)}\n\n[기존 요약]\n${c.summary}`;
  const started = Date.now();
  const res = await callBackgroundMemory(SYSTEM, [{ role: "user", content: user }], undefined, "semantic-verify-probe", {
    modelId: "gemini-3.8-flash",
    temperature: 0,
    maxTokens: 900,
  });
  const usage = res.usage as any;
  let parsed: any = null;
  let parseError: string | null = null;
  try { parsed = parseJson(res.text); } catch (e) { parseError = String((e as Error).message); }
  return {
    id: c.id,
    window: c.window,
    summary: c.summary,
    dialogue: dialogue(c.window),
    ms: Date.now() - started,
    model: "gemini-3.8-flash",
    usage: {
      inputTokens: usage.inputTokens,
      outputTokens: usage.outputTokens,
      billed: usage.cheaperInferenceBilledCostUsd ?? null,
      upstream: usage.upstreamCostUsd ?? null,
      finishReason: usage.finishReason ?? null,
      responseModelId: usage.responseModelId ?? null,
      providerRequestId: usage.providerRequestId ?? null,
    },
    rawText: res.text,
    parsed,
    parseError,
  };
}
