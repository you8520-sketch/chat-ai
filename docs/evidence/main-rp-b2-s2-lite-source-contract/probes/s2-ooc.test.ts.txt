const originalLoad = (Module as unknown as { _load: typeof Module._load })._load;
(Module as unknown as { _load: typeof Module._load })._load = function (request: string, parent: NodeModule, isMain: boolean) {
  if (request === "server-only") return {};
  return originalLoad(request, parent, isMain);
} as typeof Module._load;

function cleanup(chat: number) {
  const db = getDb();
  db.prepare("DELETE FROM episodic_memory_facts WHERE chat_id=?").run(chat);
  db.prepare("DELETE FROM chat_turn_summaries WHERE chat_id=?").run(chat);
  db.prepare("DELETE FROM chat_memories WHERE chat_id=?").run(chat);
  db.prepare("DELETE FROM messages WHERE chat_id=?").run(chat);
  db.prepare("DELETE FROM chats WHERE id=?").run(chat);
}
function seed(chat: number, turns: { user: string; assistant: string }[]) {
  cleanup(chat);
  const db = getDb();
  db.prepare("DELETE FROM users WHERE id=?").run(USER);
  db.prepare("DELETE FROM characters WHERE id=?").run(CHAR);
  db.prepare(`INSERT INTO users (id, email, nickname, pw_hash) VALUES (?,?,?,?)`).run(USER, `o-${USER}@t.l`, "렌", "x");
  db.prepare(`INSERT INTO characters (id, name) VALUES (?,?)`).run(CHAR, "라이크");
  db.prepare(`INSERT INTO chats (id, user_id, character_id, mode) VALUES (?,?,?,'safe')`).run(chat, USER, CHAR);
  getOrCreateChatMemory(chat, USER, CHAR, "pro");
  db.prepare(`INSERT INTO messages (chat_id, role, content, model) VALUES (?,?,?,?)`).run(chat, "assistant", "인사.", "greeting");
  for (const t of turns) {
    const uid = Number(db.prepare(`INSERT INTO messages (chat_id, role, content, model) VALUES (?,?,?,?)`).run(chat, "user", t.user, "user").lastInsertRowid);
    db.prepare(`INSERT INTO messages (chat_id, role, content, model, user_message_id) VALUES (?,?,?,?,?)`).run(chat, "assistant", t.assistant, "test", uid);
  }
  updateChatMemory(chat, USER, CHAR, { message_count: turns.length, membership_tier: "pro" });
  db.prepare(`INSERT INTO messages (chat_id, role, content, model, generation_status) VALUES (?,?,?,'user','submitted')`).run(chat, "user", "다음 입력");
}

const GOOD = "렌이 라이크를 아마추어 같다고 지적하자 라이크는 부족함을 인정하며 시선을 피함. 렌은 내일까지 검을 갈아 주겠다고 약속했고 라이크는 고개를 끄덕임. 서린이 문을 열고 들어와 라이크에게 목례했으며 렌이 시장에 다녀왔다고 하자 라이크는 누나가 시장에서 렌을 알아봤다고 말함.";
const OOC_ECHO = `${GOOD} 렌은 앞으로 라이크 대사를 더 짧게 쓰라고 요청함.`;
const OOC_PAST = `${GOOD} 렌과 라이크는 3년 전 용병단 동료였음.`;

async function run(name: string, turns: { user: string; assistant: string }[], lunaOut: string | null) {
  const chat = 940101;
  seed(chat, turns);
  const seen: string[] = [];
  let calls = 0;
  rolling.__setSummarizeTurnBatchCallerForTests(async (_s, history) => {
    calls += 1;
    seen.push(history.map((m) => m.content).join("\n"));
    return { text: lunaOut ?? GOOD };
  });
  const ok = await rolling.processRollingSummaryBatch({ chatId: chat, userId: USER, characterId: CHAR, charName: "라이크", tier: "pro", memoryCapacity: MEMORY_CAPACITY_FIXED, userPersona: PERSONA } as never);
  const recs = turnSummary.listMemoryRecordsForChat(chat).filter((r) => !r.inactive);
  const injectible = recs.filter((r) => turnSummary.isPromptInjectibleMemoryRecord(r));
  R[name] = {
    sealed: ok,
    lunaCalls: calls,
    oocTextReachedLuna: seen.some((s) => /\(OOC:/.test(s.split("[요약 대상 RP source 턴]")[0] ?? "")),
    lunaInputDialogue: (seen[0] ?? "").slice(0, 900),
    records: recs.map((r) => ({ start: r.turnStart, end: r.turnEnd, kind: (r as any).kind ?? null, scopes: (r as any).scopes ?? null, text: r.summary.slice(0, 160) })),
    injectibleCount: injectible.length,
    globalRebuild: turnSummary.rebuildLorebookFromRecords(chat),
  };
}

  await run("mixed_plain_ooc_luna_echoes_ooc", [...NORMAL.slice(0, 3), OOC_PLAIN, NORMAL[3]], OOC_ECHO);
  await run("mixed_bait_ooc_luna_good", [...NORMAL.slice(0, 3), OOC_BAIT, NORMAL[3]], null);
  await run("mixed_bait_ooc_luna_adopts_past", [...NORMAL.slice(0, 3), OOC_BAIT, NORMAL[3]], OOC_PAST);
  await run("ooc_only_5_plain", Array.from({ length: 5 }, (_, i) => ({ user: `(OOC: 문체를 더 짧게 해 줘 ${i + 1}.)`, assistant: `(OOC 확인 ${i + 1}.)` })), null);
  await run("ooc_only_5_hardstop", Array.from({ length: 5 }, (_, i) => ({ user: `(OOC: 앞으로 롤플레잉 중단하고 설정 얘기하자 ${i + 1})`, assistant: `(OOC 확인 ${i + 1}.)` })), null);
  await run("mixed_hardstop_ooc_good_luna", [...NORMAL.slice(0, 3), { user: "(OOC: 앞으로 롤플레잉 중단하고 설정 얘기하자)", assistant: "(OOC 확인.)" }, NORMAL[3]], null);
});
