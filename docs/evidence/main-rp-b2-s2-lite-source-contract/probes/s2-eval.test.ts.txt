function dialogueOf(w: Win): string {

type Item = { actor?: unknown; kind?: unknown; turn?: unknown; sourceRole?: unknown; quote?: unknown; relationshipState?: unknown; eventStatus?: unknown };
type Parsed =
  | { ok: true; recap: string; items: Item[]; extra: string[] }
  | { ok: false; reason: "blank" | "no_json" | "malformed_json" | "missing_recap" | "missing_items" };

function parseS2(raw: string): Parsed {
  const text = raw.trim();
  if (!text) return { ok: false, reason: "blank" };
  const fenced = text.replace(/^```(?:json)?\s*/i, "").replace(/\s*```$/i, "");
  const a = fenced.indexOf("{");
  const b = fenced.lastIndexOf("}");
  if (a < 0 || b <= a) return { ok: false, reason: "no_json" };
  let obj: { recap?: unknown; items?: unknown };
  try {
    obj = JSON.parse(fenced.slice(a, b + 1)) as typeof obj;
  } catch {
    return { ok: false, reason: "malformed_json" };
  }
  if (typeof obj.recap !== "string" || !obj.recap.trim()) return { ok: false, reason: "missing_recap" };
  if (!Array.isArray(obj.items)) return { ok: false, reason: "missing_items" };
  const extra: string[] = [];
  if (fenced.slice(0, a).trim() || fenced.slice(b + 1).trim()) extra.push("text_outside_json");
  return { ok: true, recap: obj.recap.replace(/\s+/g, " ").trim(), items: obj.items as Item[], extra };
}

const ENUM = {
  actor: ["user", "character", "npc"],
  kind: ["event", "claim"],
  sourceRole: ["user", "assistant"],
  relationshipState: ["none", "proposal", "accepted", "declined", "uncertain"],
  eventStatus: ["active", "cancelled", "historical", "none"],
} as const;

type Verdict = {
  actor: string;
  kind: string;
  turn: number | null;
  sourceRole: string;
  quote: string;
  relationshipState: string;
  eventStatus: string;
  schemaOk: boolean;
  schemaProblems: string[];
  accepted: boolean;
  reason: string | null;
  authority: string | null;
  quoteInOppositeRole: boolean;
  rendered: string | null;
};

function verifyItem(w: Win, it_: Item): Verdict {
  const s = (v: unknown) => (typeof v === "string" ? v : "");
  const actor = s(it_.actor);
  const kind = s(it_.kind);
  const sourceRole = s(it_.sourceRole);
  const quote = s(it_.quote);
  const relationshipState = s(it_.relationshipState);
  const eventStatus = s(it_.eventStatus);
  const turn = Number.isInteger(it_.turn) ? (it_.turn as number) : null;
  const problems: string[] = [];
  for (const k of Object.keys(ENUM) as (keyof typeof ENUM)[]) {
    const val = s((it_ as Record<string, unknown>)[k]);
    if (!(ENUM[k] as readonly string[]).includes(val)) problems.push(`${k}=${JSON.stringify((it_ as Record<string, unknown>)[k])}`);
  }
  if (turn == null) problems.push("turn");
  if (!quote) problems.push("quote");
  const base: Verdict = {
    actor, kind, turn, sourceRole, quote, relationshipState, eventStatus,
    schemaOk: problems.length === 0, schemaProblems: problems,
    accepted: false, reason: null, authority: null, quoteInOppositeRole: false, rendered: null,
  };
  const idx = turn == null ? -1 : turn - w.start;
  const t = idx >= 0 && idx < w.turns.length ? w.turns[idx]! : null;
  if (!t || (sourceRole !== "user" && sourceRole !== "assistant")) {
    return { ...base, reason: t ? "role_invalid" : "turn_out_of_window" };
  }
  const userText = normalizeEvidenceSpan(t.user);
  const asstText = normalizeEvidenceSpan(t.assistant);
  const nq = normalizeEvidenceSpan(quote);
  base.quoteInOppositeRole = sourceRole === "user" ? asstText.includes(nq) : userText.includes(nq);
  const userId = (turn as number) * 2 - 1;
  const sources = {
    user: { id: userId, text: userText },
    assistant: { id: userId + 1, text: asstText },
  };
  const evidenceType: EpisodicExtractedFact["evidence_type"] =
    sourceRole === "user" ? "explicit_user_statement" : kind === "claim" ? "explicit_character_claim" : "explicit_scene_event";
  const subject = actor === "user" ? "user" : actor === "npc" ? "npc" : "rike";
  const fact: EpisodicExtractedFact = {
    category: relationshipState !== "none" && relationshipState !== "" ? "relationship" : "character",
    subject,
    attribute: "scene_event",
    value: "x",
    importance: "normal",
    fact_text: quote,
    evidence_type: evidenceType,
    evidence: { source_role: sourceRole, source_quote: quote },
  };
  const d = evaluateEpisodicFactEvidence(fact, sources, { userPersona: PERSONA });
  if (d.accepted) {
    return { ...base, accepted: true, authority: d.evidence.authority, rendered: renderVerifiedEvidenceText(d.evidence) };
  }
  return { ...base, reason: d.reason };
}

const CONTENT_TOKEN = /[\p{L}\p{N}]{2,}/gu;
const STOP = new Set(["렌", "라이크", "함", "음", "임", "하자", "하고", "하며", "했다", "이후", "그리고", "라이크는", "렌은", "렌이", "라이크가"]);
function tokens(s: string): Set<string> {
  const out = new Set<string>();
  for (const m of s.normalize("NFC").match(CONTENT_TOKEN) ?? []) {
    const t = m.replace(/(은|는|이|가|을|를|에|의|과|와|도|로|으로|에서|하며|하고|하자|했다|했고|하는|인|한)$/u, "");
    if (t.length >= 2 && !STOP.has(t) && !STOP.has(m)) out.add(t);
  }
  return out;
}
function sentencesOf(recap: string): string[] {
  return recap.split(/(?<=[.。!?])\s+|(?<=함\.)\s+/u).map((x) => x.trim()).filter(Boolean);
}
/** Lexical overlap is used ONLY to measure what a quote-only projection would drop; it never grants authority. */
function coveredByQuotes(sentence: string, quotes: string[], min = 2): boolean {
  const st = tokens(sentence);
  for (const q of quotes) {
    const qt = tokens(q);
    let n = 0;
    for (const t of st) if ([...qt].some((x) => x === t || x.includes(t) || t.includes(x))) n += 1;
    if (n >= min) return true;
  }
  return false;
}

function guardOf(recap: string, w: Win) {
  const validate = validateSummaryNarrative(recap, "main_canon");
  const grounded = validate.ok ? isRollingSummaryGroundedInDialogue(recap, dialogueOf(w), PERSONA, CHAR) : false;
  return { chars: recap.length, validateOk: validate.ok, validateReason: validate.ok ? null : (validate as { reason?: string }).reason ?? "invalid", grounded, sealable: validate.ok && grounded };
}

it("E1-E5 real Luna F vs S2-lite", () => {
  const perCall: unknown[] = [];
  const usage = { F: { calls: 0, inTok: 0, outTok: 0, usd: 0, ms: 0 }, S: { calls: 0, inTok: 0, outTok: 0, usd: 0, ms: 0 } };
  const parseFail: Record<string, number> = {};
  let sCalls = 0;
  let sParsed = 0;
  const goldAgg: Record<string, { pass: number; total: number; na: number }> = {};
  for (const r of raws.sort((a, b) => (a.id + a.arm + a.rep).localeCompare(b.id + b.arm + b.rep))) {
    const w = winById.get(r.id)!;
    const u = usage[r.arm];
    u.calls += 1;
    u.inTok += r.usage?.inputTokens ?? 0;
    u.outTok += r.usage?.outputTokens ?? 0;
    u.usd += r.billedUsd ?? 0;
    u.ms += r.ms ?? 0;
    if (r.error || r.text == null) {
      perCall.push({ id: r.id, arm: r.arm, rep: r.rep, providerError: r.error ?? "no_text" });
      continue;
    }
    if (r.arm === "F") {
      const recap = r.text.replace(/\s+/g, " ").trim();
      perCall.push({ id: r.id, arm: "F", rep: r.rep, group: w.group, recap, guard: guardOf(recap, w), finishReason: r.usage?.finishReason ?? null, outTok: r.usage?.outputTokens ?? null });
      continue;
    }
    sCalls += 1;
    const p = parseS2(r.text);
    if (!p.ok) {
      parseFail[p.reason] = (parseFail[p.reason] ?? 0) + 1;
      perCall.push({ id: r.id, arm: "S", rep: r.rep, group: w.group, parseFailure: p.reason, finishReason: r.usage?.finishReason ?? null, outTok: r.usage?.outputTokens ?? null, rawHead: r.text.slice(0, 200) });
      continue;
    }
    sParsed += 1;
    const verdicts = p.items.map((i) => verifyItem(w, i));
    const verified = verdicts.filter((v) => v.accepted);
    const quotes = verified.map((v) => v.quote);
    const goldRes: Record<string, boolean | "n/a"> = {};
    for (const g of goldFor(w)) {
      const res = g.test(verified);
      goldRes[g.key] = res;
      const a = (goldAgg[g.key] ??= { pass: 0, total: 0, na: 0 });
      if (res === "n/a") a.na += 1;
      else {
        a.total += 1;
        if (res) a.pass += 1;
      }
    }
    const sentences = sentencesOf(p.recap);
    const covered = sentences.map((s) => coveredByQuotes(s, quotes));
    const terms = RECAP_TERMS[w.id] ?? [];
    const termHits = terms.map((t) => ({ term: t, recap: new RegExp(t).test(p.recap), items: verified.some((v) => new RegExp(t).test(v.quote)) }));
    perCall.push({
      id: r.id, arm: "S", rep: r.rep, group: w.group,
      finishReason: r.usage?.finishReason ?? null, outTok: r.usage?.outputTokens ?? null, inTok: r.usage?.inputTokens ?? null,
      recap: p.recap, recapGuard: guardOf(p.recap, w), extra: p.extra,
      itemCount: verdicts.length,
      itemsSchemaOk: verdicts.filter((v) => v.schemaOk).length,
      itemsVerified: verified.length,
      rejections: verdicts.filter((v) => !v.accepted).map((v) => ({ reason: v.reason, quote: v.quote.slice(0, 60), role: v.sourceRole, turn: v.turn })),
      authorityCounts: verified.reduce<Record<string, number>>((m, v) => ((m[v.authority!] = (m[v.authority!] ?? 0) + 1), m), {}),
      roleMislabeled: verdicts.filter((v) => !v.accepted && v.quoteInOppositeRole).length,
      actorRoleMismatch: verdicts.filter((v) => (v.actor === "user") !== (v.sourceRole === "user")).length,
      items: verdicts.map((v) => ({ actor: v.actor, kind: v.kind, turn: v.turn, role: v.sourceRole, rel: v.relationshipState, status: v.eventStatus, accepted: v.accepted, authority: v.authority, reason: v.reason, quote: v.quote })),
      renderedQuotes: verified.map((v) => v.rendered),
      renderedChars: verified.reduce((n, v) => n + (v.rendered?.length ?? 0), 0),
      recapSentences: sentences.length,
      recapSentencesQuoteCovered: covered.filter(Boolean).length,
      uncoveredSentences: sentences.filter((_, i) => !covered[i]),
      gold: goldRes,
      termHits,
    });
  }
  R.usage = usage;
  R.sCalls = sCalls;
  R.sParsed = sParsed;
  R.parseFail = parseFail;
  R.goldAgg = goldAgg;
  R.perCall = perCall;
});

  if (!w) return;
  const out = raws.filter((r) => r.id === "OOC_main_class" && r.text).map((r) => {
    if (r.arm === "F") return { arm: "F", rep: r.rep, recap: r.text!.replace(/\s+/g, " ").trim() };
    const p = parseS2(r.text!);
    if (!p.ok) return { arm: "S", rep: r.rep, parseFailure: p.reason };
    const vs = p.items.map((i) => verifyItem(w, i));
    return { arm: "S", rep: r.rep, recap: p.recap, items: vs.map((v) => ({ actor: v.actor, kind: v.kind, role: v.sourceRole, accepted: v.accepted, authority: v.authority, quote: v.quote })) };
  });
  R.E8_ooc = out;
});
